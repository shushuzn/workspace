import joblib

# 加载集成模型
ensemble = joblib.load('LIG_ensemble_GP_EN.pkl')

# 准备特征 (6 个)
features = ['E_Jcm2', 'v_mms', 'co_ratio', 'ssa_m2g', 'id_ig', 'P_W']
X = your_data[features].values

# 缩放
X_scaled = ensemble['scaler_6'].transform(X)

# 预测
X_3 = your_data[ensemble['features_3']].values
X_3_scaled = ensemble['scaler_3'].transform(X_3)

y_pred_gp = ensemble['gp_model'].predict(X_3_scaled)
y_pred_en = ensemble['en_model'].predict(X_6_scaled)
y_pred = ensemble['gp_weight'] * y_pred_gp + (1 - ensemble['gp_weight']) * y_pred_en
