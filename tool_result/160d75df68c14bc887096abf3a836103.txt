def cmd_all():
    config = load_config()
    feishu = load_feishu_config()
    history = load_history()
    
    all_news = fetch_all()
    if not all_news:
        warn("No news")
        return 0
    
    new_news = [n for n in all_news if n['title'] not in history]
    if not new_news:
        warn("No new news")
        return 0
    
    info(f"Total: {len(all_news)} | New: {len(new_news)}")
    print()
    
    items = new_news[:8]
    token = get_access_token()
    if not token:
        error("Failed to get token")
        return 0
    
    sent = 0
    for n in items:
        try:
            card = build_news_card(n['title'], n['url'], n.get('source', ''))
            result = send_card(token, feishu.get('default_receive_id', ''), card)
            if result.get('code') == 0:
                history.append(n['title'])
                ok(n['title'][:40] + "...")
                sent += 1
            else:
                error(f"Code {result.get('code')}")
            time.sleep(0.2)
        except Exception as e:
            error(str(e)[:50])
            break
    
    save_history(history)
    return sent
