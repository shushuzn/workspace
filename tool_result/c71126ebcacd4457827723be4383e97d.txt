    def get_token(self) -> str:
        """Get valid token, refreshing if necessary."""
        if self._token and self._expires_at:
            if datetime.now() < self._expires_at - timedelta(seconds=self.refresh_before_expiry):
                return self._token
        
        # Try load from cache
        if self.load_from_cache():
            return self._token
        
        # Refresh from API
        return self.refresh()
