class DCFModel:
    """DCF估值模型"""
    
    @staticmethod
    def calculate(
        fcf_per_share: float,
        growth_rate: float,
        terminal_growth: float = 0.03,
        discount_rate: float = 0.10,
        years: int = 5,
        shares: float = 1e9
    ) -> Tuple[float, float, float]:
        """
        DCF三场景估值
        
        Returns:
            (base, bull, bear)
        """
        # 悲观场景
        g_bear = growth_rate * 0.5
        dr_bear = discount_rate + 0.02
        bear = DCFModel._dcf_value(fcf_per_share, g_bear, terminal_growth, dr_bear, years, shares)
        
        # 基准场景
        base = DCFModel._dcf_value(fcf_per_share, growth_rate, terminal_growth, discount_rate, years, shares)
        
        # 乐观场景
        g_bull = growth_rate * 1.5
        dr_bull = discount_rate - 0.02
        bull = DCFModel._dcf_value(fcf_per_share, g_bull, terminal_growth, dr_bull, years, shares)
        
        return bear, base, bull
