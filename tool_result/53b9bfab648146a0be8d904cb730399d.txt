"""REST API server for Stock PRO"""
import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs

from .core import analyze, analyze_multiple
from .screener import StockScreener
from .portfolio import PortfolioManager

PORT = 8765

class APIHandler(BaseHTTPRequestHandler):
    def log_message(self, format, *args):
        pass  # Suppress logging
    
    def send_json(self, data, status=200):
        self.send_response(status)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Access-Control-Allow-Origin', '*')
        self.end_headers()
        self.wfile.write(json.dumps(data).encode('utf-8'))
    
    def do_GET(self):
        parsed = urlparse(self.path)
        path = parsed.path
        query = parse_qs(parsed.query)
        
        if path == '/':
            from datetime import datetime
            self.send_json({"service": "Stock PRO API", "version": "11.0", "timestamp": datetime.now().isoformat(), "endpoints": ["/analyze/<symbol>", "/compare?symbols=NVDA,META", "/screener", "/portfolio"]})
        
        elif path == '/analyze':
            symbol = query.get('symbol', [''])[0]
            if symbol:
                try:
                    live = query.get('live', ['false'])[0].lower() == 'true'
                    data = analyze(symbol, live=live)
                    self.send_json(data)
                except Exception as e:
                    self.send_json({"error": str(e)}, 400)
            else:
                self.send_json({"error": "Missing symbol parameter"}, 400)
        
        elif path == '/compare':
            symbols = query.get('symbols', [''])[0].split(',')
            if symbols and symbols[0]:
                try:
                    results = analyze_multiple(symbols)
                    self.send_json({"results": results})
                except Exception as e:
                    self.send_json({"error": str(e)}, 400)
            else:
                self.send_json({"error": "Missing symbols parameter"}, 400)
        
        elif path == '/screener':
            try:
                min_score = int(query.get('min_score', [60])[0])
                min_upside = float(query.get('min_upside', [15])[0])
                max_pe = float(query.get('max_pe', [40])[0])
                s = StockScreener(min_score=min_score, min_upside=min_upside, max_pe=max_pe)
                results = s.screen()
                self.send_json({"results": results, "count": len(results)})
            except Exception as e:
                self.send_json({"error": str(e)}, 400)
        
        elif path == '/portfolio':
            try:
                pm = PortfolioManager()
                self.send_json({"positions": pm.positions})
            except Exception as e:
                self.send_json({"error": str(e)}, 400)
        
        elif path == '/health':
            from datetime import datetime
            self.send_json({"status": "ok", "timestamp": datetime.now().isoformat()})
        
        else:
            self.send_json({"error": "Not found"}, 404)
    
    def do_POST(self):
        parsed = urlparse(self.path)
        path = parsed.path
        content_length = int(self.headers.get('Content-Length', 0))
        
        if path == '/portfolio/add':
            try:
                body = json.loads(self.rfile.read(content_length))
                pm = PortfolioManager()
                result = pm.add(body['symbol'], body['shares'], body['cost'])
                self.send_json({"result": result})
            except Exception as e:
                self.send_json({"error": str(e)}, 400)
        
        else:
            self.send_json({"error": "Not found"}, 404)


def start_server(port=PORT):
    """Start API server"""
    server = HTTPServer(('localhost', port), APIHandler)
    print(f"[API] Stock PRO API running on http://localhost:{port}")
    print("[API] Endpoints:")
    print("  GET  /                    - API info")
    print("  GET  /analyze?symbol=NVDA - Single stock analysis")
    print("  GET  /compare?symbols=NVDA,META,JPM - Multiple stocks")
    print("  GET  /screener?min_score=60 - Stock screener")
    print("  GET  /portfolio           - Portfolio")
    print("  POST /portfolio/add       - Add position")
    print("  GET  /health              - Health check")
    print()
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\n[API] Server stopped")
        server.shutdown()
