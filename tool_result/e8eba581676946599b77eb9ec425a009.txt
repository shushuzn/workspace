def cmd_status(args):
    data = NewsData()
    total = len(data.data["news"])
    unsent = len([n for n in data.data["news"] if not n.get("sent")])
    sent = total - unsent
    
    # 统计
    sectors = {}
    sentiments = {"利好": 0, "利空": 0, "中性": 0}
    for n in data.data["news"]:
        sectors[n.get("sector","通用")] = sectors.get(n.get("sector","通用"), 0) + 1
        sentiments[n.get("sentiment","中性")] = sentiments.get(n.get("sentiment","中性"), 0) + 1
    
    # 格式化
    now = datetime.now().strftime("%m-%d %H:%M")
    sent_pct = sent / total * 100 if total > 0 else 0
    
    # 移动端优化
    print(f"\n{c(Colors.header(), '=== STATUS @ ' + now + ' ===')}\n")
    
    # 进度条
    bar_len = 16
    filled = int(sent_pct / 100 * bar_len)
    bar = c(Colors.GREEN, "#" * filled) + "-" * (bar_len - filled)
    print(f"Progress: [{bar}] {sent_pct:.0f}%")
    print(f"News: {total} | Sent: {sent} | Unsent: {unsent}")
    print(f"Push: QQ={data.data['stats'].get('qq',0)} FS={data.data['stats'].get('feishu',0)}")
    
    # 情感
    print(f"Sentiment: {cp('+' + str(sentiments['利好']))} {cn('-' + str(sentiments['利空']))} {cne('=' + str(sentiments['中性']))}")
    
    # 风险
    risk_score = sentiments["利空"] * 2 - sentiments["利好"]
    if risk_score > 10:
        risk_text = cn("HIGH")
    elif risk_score > 5:
        risk_text = c(Colors.YELLOW, "MEDIUM")
    else:
        risk_text = cp("LOW")
    print(f"\nRisk: {risk_text} (score: {risk_score:+d})")
    print()
