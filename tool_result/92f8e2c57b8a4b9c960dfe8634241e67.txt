#!/usr/bin/env python3
"""News Collector v1.0 - SQLite backed real-time news"""
import sqlite3, urllib.request, re
from datetime import datetime
from pathlib import Path

DB = Path(__file__).parent / "news_cache.db"
KW = {"NVDA": ["NVIDIA", "英伟达", "算力"], "TSLA": ["Tesla", "特斯拉", "马斯克"], "AI": ["AI", "人工智能"]}

def init():
    c = sqlite3.connect(DB).cursor()
    c.execute("CREATE TABLE IF NOT EXISTS n(id INTEGER PRIMARY KEY, s TEXT, t TEXT, u TEXT, ts TEXT)")
    c.execute("CREATE TABLE IF NOT EXISTS src(n TEXT PRIMARY KEY, ts TEXT)")
    sqlite3.connect(DB).commit()

def save(news):
    c = sqlite3.connect(DB).cursor()
    for n in news:
        if not c.execute("SELECT 1 FROM n WHERE t=? AND s=?", (n["t"], n["s"])).fetchone():
            c.execute("INSERT INTO n(s,t,u,ts) VALUES(?,?,?,?)", (n["s"], n["t"], n["u"], n["ts"]))
    sqlite3.connect(DB).commit()

def get(sym=None, h=24, lim=50):
    c = sqlite3.connect(DB).cursor()
    if sym:
        kws = KW.get(sym.upper(), [sym])
        sql = " OR ".join(["t LIKE ?" for _ in kws])
        p = [f"%{k}%" for k in kws] + [h, lim]
        r = c.execute(f"SELECT s,t,u,ts FROM n WHERE ({sql}) AND ts>datetime('now','-{h} hours') ORDER BY ts DESC LIMIT ?", p)
    else:
        r = c.execute("SELECT s,t,u,ts FROM n WHERE ts>datetime('now','-24 hours') ORDER BY ts DESC LIMIT ?", (lim,))
    return [{"source": x[0], "title": x[1], "url": x[2], "time": x[3]} for x in r.fetchall()]

def fetch():
    news = []
    for url, src, pat in [
        ("https://finance.sina.com.cn/", "sina", r'href="(https?://finance\.sina\.com\.cn/[^"]+)"[^>]*>\s*([^<]{15,150})\s*<'),
        ("https://xnews.jin10.com/", "jin10", r'href="(https://xnews\.jin10\.com/details/\d+)"[^>]*>(.*?)</a>'),
    ]:
        try:
            req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
            with urllib.request.urlopen(req, timeout=15) as r:
                html = r.read().decode("utf-8", errors="ignore")
            for m in re.findall(pat, html, re.DOTALL if "jin10" in src else 0):
                title = m[1].strip() if len(m) > 1 else m[0]
                if "jin10" in src:
                    title = re.sub("<[^>]+>", "", title)[:200]
                if len(title) > 15:
                    news.append({"s": src, "t": title, "u": m[0], "ts": datetime.now().isoformat()})
        except Exception as e:
            print(f"{src}: {e}")
    return news

if __name__ == "__main__":
    print(f"[{datetime.now().strftime('%H:%M:%S')}] Fetching...")
    init()
    news = fetch()
    save(news)
    print(f"[{datetime.now().strftime('%H:%M:%S')}] Saved {len(news)} items")
    for n in get()[:5]:
        print(f"  [{n['source']}] {n['title'][:50]}...")
