def cmd_digest(args):
    data = NewsData()
    days = int(args.days) if hasattr(args, 'days') else 1
    cutoff = datetime.now() - timedelta(days=days)
    recent = [n for n in data.data["news"] if datetime.fromisoformat(n.get("fetched_at","2000")).replace(tzinfo=None) > cutoff]
    
    if not recent:
        print(f"No news in last {days} day(s)")
        return
    
    now = datetime.now().strftime("%Y-%m-%d %H:%M")
    
    # 统计
    sentiments = {"利好": [], "利空": [], "中性": []}
    for n in recent:
        sentiments[n.get("sentiment","中性")].append(n)
    
    # 板块统计
    sectors = {}
    for n in recent:
        sectors[n.get("sector","通用")] = sectors.get(n.get("sector","通用"), 0) + 1
    top_sectors = sorted(sectors.items(), key=lambda x: -x[1])[:5]
    
    # 来源统计
    sources = {}
    for n in recent:
        src = SOURCE_MAP.get(n.get('source',''), n.get('category','Other'))
        sources[src] = sources.get(src, 0) + 1
    top_sources = sorted(sources.items(), key=lambda x: -x[1])[:5]
    
    # 关键词统计
    keywords = {}
    for n in recent:
        for kw in n.get('keywords', []):
            keywords[kw] = keywords.get(kw, 0) + 1
    top_keywords = sorted(keywords.items(), key=lambda x: -x[1])[:8]
    
    # 风险评估
    risk_score = len(sentiments["利空"]) * 2 - len(sentiments["利好"])
    if risk_score > 10:
        risk = "HIGH"
    elif risk_score > 5:
        risk = "MEDIUM"
    else:
        risk = "LOW"
    
    total = len(recent)
    pos = len(sentiments['利好'])
    neg = len(sentiments['利空'])
    neu = len(sentiments['中性'])
    
    print(f"""
+=== NEWS DIGEST @ {now} =============================+
|  Total: {total:>3} news  [+利好]{pos:>2}  [-利空]{neg:>2}  [=中性]{neu:>3}  |
+========================================================+
|  RISK: {risk:<6}  Score: {risk_score:>+3}                            |
+--------------------------------------------------------+
|  TOP SECTORS:                                          |""")
    
    for sec, cnt in top_sectors[:4]:
        pct = cnt / total * 100
        bar = "#" * int(pct / 5) + "-" * (20 - int(pct / 5))
        print(f"|    {sec:<6} [{bar}] {cnt:>3} ({pct:>4.1f}%)    |")
    
    print(f"""+--------------------------------------------------------+
|  SOURCES: {', '.join([f'{s[0]}({s[1]})' for s in top_sources[:4]])}    |
+--------------------------------------------------------+
|  HOT TAGS: {' '.join([f'#{s[0]}' for s in top_keywords[:6]])}  |
+========================================================+
""")

    # 情感详情 - 紧凑显示
    for sent_type, news_list in sentiments.items():
        if not news_list:
            continue
        icon = "[+]" if sent_type == "利好" else "[-]" if sent_type == "利空" else "[=]"
        print(f"{icon} {sent_type} ({len(news_list)} items)")
