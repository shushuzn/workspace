// Idle Empire - Boss 战斗系统 v0.6

const BOSSES = [
    { id: 'goblin', name: '哥布林首领', icon: '👺', hp: 100, reward: 500, unlockGold: 1000, damage: 1 },
    { id: 'troll', name: '洞穴巨魔', icon: '🧌', hp: 1000, reward: 5000, unlockGold: 10000, damage: 5 },
    { id: 'dragon', name: '远古巨龙', icon: '🐉', hp: 10000, reward: 50000, unlockGold: 100000, damage: 25 },
    { id: 'demon', name: '深渊恶魔', icon: '👿', hp: 100000, reward: 500000, unlockGold: 1000000, damage: 100 },
    { id: 'titan', name: '泰坦巨人', icon: '👽', hp: 1000000, reward: 5000000, unlockGold: 10000000, damage: 500 }
];

var currentBoss = null;
var bossHp = 0;
var bossMaxHp = 0;
var bossDamageTimer = null;

function initBossSystem() {
    if (!G) return;
    
    // 检查解锁
    var unlockedBosses = BOSSES.filter(function(b) {
        return G.totalEarned >= b.unlockGold;
    });
    
    if (unlockedBosses.length === 0) return;
    
    // 随机选择或继续当前Boss
    if (!G.currentBoss || !BOSSES.find(function(b) { return b.id === G.currentBoss; })) {
        G.currentBoss = unlockedBosses[0].id;
    }
    
    if (!G.bossHp) G.bossHp = getBossById(G.currentBoss).hp;
    if (!G.bossMaxHp) G.bossMaxHp = getBossById(G.currentBoss).hp;
    
    currentBoss = getBossById(G.currentBoss);
    bossHp = G.bossHp;
    bossMaxHp = G.bossMaxHp;
    
    renderBoss();
    startBossDamage();
}

function getBossById(id) {
    for (var i = 0; i < BOSSES.length; i++) {
        if (BOSSES[i].id === id) return BOSSES[i];
    }
    return null;
}

function renderBoss() {
    var container = document.getElementById('boss-container');
    if (!container || !currentBoss || !G) return;
    
    var hpPercent = (bossHp / bossMaxHp) * 100;
    var unlocked = BOSSES.filter(function(b) { return G.totalEarned >= b.unlockGold; });
    var nextBoss = unlocked.find(function(b) { return BOSSES.indexOf(b) > BOSSES.indexOf(currentBoss); });
    
    container.innerHTML = '<div class="boss-section">' +
        '<div class="boss-header">' +
            '<div class="boss-info">' +
                '<span class="boss-icon">' + currentBoss.icon + '</span>' +
                '<div>' +
                    '<div class="boss-name">' + currentBoss.name + '</div>' +
                    '<div class="boss-damage-info">每10秒 -' + formatNumber(currentBoss.damage) + ' HP</div>' +
                '</div>' +
            '</div>' +
        '</div>' +
        '<div class="boss-hp">' +
            '<div class="boss-hp-bar">' +
                '<div class="boss-hp-fill" style="width:' + hpPercent + '%"></div>' +
            '</div>' +
            '<div class="boss-hp-text">' + formatNumber(bossHp) + ' / ' + formatNumber(bossMaxHp) + '</div>' +
        '</div>' +
        '<div class="boss-reward">击杀奖励: ' + formatNumber(currentBoss.reward) + ' 💰</div>' +
    '</div>';
}

function startBossDamage() {
    if (bossDamageTimer) clearInterval(bossDamageTimer);
    
    bossDamageTimer = setInterval(function() {
        if (!G || !currentBoss) return;
        
        // 每10秒扣血
        bossHp -= currentBoss.damage;
        G.bossHp = bossHp;
        
        if (bossHp <= 0) {
            // Boss被击杀
            defeatBoss();
        } else {
            renderBoss();
        }
    }, 10000);
}

function defeatBoss() {
    if (!G || !currentBoss) return;
    
    // 发放奖励
    G.gold += currentBoss.reward;
    G.totalEarned += currentBoss.reward;
    G.bossesDefeated = (G.bossesDefeated || 0) + 1;
    
    showMsg('🎉 击杀 ' + currentBoss.name + '! +' + formatNumber(currentBoss.reward) + ' 金币!', 'success');
    
    // 下一个Boss
    var unlocked = BOSSES.filter(function(b) { return G.totalEarned >= b.unlockGold; });
    var idx = BOSSES.indexOf(currentBoss);
    
    if (idx < BOSSES.length - 1 && unlocked.length > idx + 1) {
        currentBoss = BOSSES[idx + 1];
        bossHp = currentBoss.hp;
        bossMaxHp = currentBoss.hp;
        G.currentBoss = currentBoss.id;
        G.bossHp = bossHp;
        G.bossMaxHp = bossMaxHp;
    } else {
        // 最后一个Boss，重置HP
        bossHp = currentBoss.hp;
        bossMaxHp = currentBoss.hp;
        G.bossHp = bossHp;
        G.bossMaxHp = bossMaxHp;
    }
    
    renderBoss();
    updateUI();
    renderStats();
    checkAchievements();
}

function getBossHpPercent() {
    return bossMaxHp > 0 ? (bossHp / bossMaxHp) * 100 : 0;
}
