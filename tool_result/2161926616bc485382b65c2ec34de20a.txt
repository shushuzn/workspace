# 情感分析
POSITIVE = ["涨", "升", "利多", "利好", "上涨", "突破", "牛市", "超预期", "增长"]
NEGATIVE = ["跌", "降", "利空", "下跌", "崩盘", "熊市", "低于预期", "萎缩"]

def analyze_sentiment(title):
    t = title.lower()
    pos = sum(1 for w in POSITIVE if w in t)
    neg = sum(1 for w in NEGATIVE if w in t)
    if pos > neg:
        return "positive", 20
    elif neg > pos:
        return "negative", -20
    return "neutral", 0

def detect_hot_topics(news_list, top_n=5):
    stopwords = {"的", "了", "在", "是", "和", "就", "不", "都", "一", "上", "也", "很", "到", "说"}
    word_freq = {}
    for n in news_list:
        words = re.findall(r'[\u4e00-\u9fff]{2,4}', n.get("title", ""))
        for w in words:
            if w not in stopwords and len(w) >= 2:
                word_freq[w] = word_freq.get(w, 0) + 1
    return sorted(word_freq.items(), key=lambda x: -x[1])[:top_n]

def cluster_news(news_list, max_clusters=5):
    hot = detect_hot_topics(news_list, top_n=max_clusters)
    clusters, used = [], set()
    for word, _ in hot:
        cluster = []
        for i, n in enumerate(news_list):
            if i in used:
                continue
            if word in n.get("title", ""):
                cluster.append(n)
                used.add(i)
        if len(cluster) >= 2:
            clusters.append({"topic": word, "news": cluster, "count": len(cluster)})
    return clusters
