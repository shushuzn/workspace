def enrich_news(news_list):
    """丰富新闻数据：分类 + 情感 + 重要性"""
    for news in news_list:
        categorize(news)
        analyze_sentiment(news)

IMPORTANCE_KEYWORDS = {
    "critical": ["战争", "核", "制裁", "衰退", "崩盘", "违约", "危机", "911", "黑天鹅"],
    "high": ["美联储", "加息", "降息", "央行", "G20", "峰会", "协议", "突破", "新高", "新低"],
    "medium": ["财报", "GDP", "CPI", "非农", "就业", "通胀", "经济", "增长"],
    "low": ["评论", "分析", "预计", "可能", "或者"],
}

def score_importance(news):
    """计算新闻重要性分数"""
    title = news.get('title', '')
    score = 0
    
    # 关键词匹配
    for level, keywords in IMPORTANCE_KEYWORDS.items():
        for kw in keywords:
            if kw in title:
                if level == 'critical': score += 10
                elif level == 'high': score += 5
                elif level == 'medium': score += 3
                else: score += 1
