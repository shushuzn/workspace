#!/usr/bin/env python3
"""
NewsHub - 新闻聚合推送入口
"""
import sys
import subprocess
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8')

BASE_DIR = Path(__file__).parent

# 颜色
CYAN = '\033[36m'
GREEN = '\033[32m'
YELLOW = '\033[33m'
RESET = '\033[0m'
BOLD = '\033[1m'


def header(text):
    print(f"\n{BOLD}{CYAN}{'─' * 50}{RESET}")
    print(f"{BOLD}{CYAN}  {text}{RESET}")
    print(f"{BOLD}{CYAN}{'─' * 50}{RESET}\n")


def run(script, *args):
    cmd = [sys.executable, BASE_DIR / script, *args]
    return subprocess.run(cmd).returncode == 0


def main():
    cmd = sys.argv[1] if len(sys.argv) > 1 else "help"
    
    # 解析额外参数
    extra_args = sys.argv[3:] if len(sys.argv) > 3 else []
    
    if cmd == "feishu":
        mode = sys.argv[2] if len(sys.argv) > 2 else "all"
        header("Feishu Push")
        run("news_feishu.py", mode, *extra_args)
    
    elif cmd == "qq":
        mode = sys.argv[2] if len(sys.argv) > 2 else "status"
        header("QQ Push")
        run("news_to_qq.py", mode)
    
    elif cmd == "fetch":
        header("Fetch News")
        run("news_fetcher.py", *extra_args)
    
    elif cmd == "test":
        header("Test All")
        print(f"{YELLOW}Testing Feishu...{RESET}\n")
        run("news_feishu.py", "all", *extra_args)
        print(f"\n{YELLOW}Testing QQ...{RESET}\n")
        run("news_to_qq.py", "status")
    
    elif cmd == "quick":
        header("Quick Mode")
        run("news_fetcher.py", "--quick")
    
    elif cmd == "export":
        header("Export News")
        run("news_fetcher.py", *sys.argv[2:])
    
    elif cmd == "server":
        header("RSS Server")
        import subprocess
        subprocess.Popen([sys.executable, Path(__file__).parent / "rss_server.py"])
        print(f"  {GREEN}✓{RESET} Server started")
    
    elif cmd == "report":
        header("Generate Report")
        run("news_report.py")
    
    elif cmd == "db":
        header("Database")
        run("news_db.py")
    
    elif cmd == "trends":
        header("Trend Analysis")
        run("news_trends.py", *sys.argv[2:])
    
    elif cmd == "sub":
        header("Subscriptions")
        run("news_subscriptions.py", *sys.argv[2:])
    
    elif cmd == "watch":
        header("Watch Mode")
        from news_fetcher import fetch_all
        from news_subscriptions import cmd_check
        news = fetch_all(quiet=True)
        cmd_check(news)
    
    elif cmd == "help":
        header("NewsHub Help")
        print(f"""
  {BOLD}用法:{RESET}
    python news_hub.py <command> [options]

  {BOLD}命令:{RESET}
    {GREEN}fetch{RESET}     抓取新闻（不推送）
    {GREEN}feishu{RESET}    飞书推送
    {GREEN}qq{RESET}        QQ推送
    {GREEN}test{RESET}     测试所有
    {GREEN}quick{RESET}    快速模式
    {GREEN}server{RESET}   启动RSS服务
    {GREEN}export{RESET}   导出新闻(JSON/CSV)
    {GREEN}report{RESET}    生成每日简报
    {GREEN}db{RESET}        数据库管理
    {GREEN}trends{RESET}    趋势分析
    {GREEN}sub{RESET}       订阅管理
    {GREEN}watch{RESET}     关键词监控
    {GREEN}help{RESET}     显示帮助

  {BOLD}飞书子命令:{RESET}
    compact    紧凑视图（纯文本）
    preview    预览新闻（不发送）
    digest     摘要推送（6条）
    alert      关键词告警
    all        全部推送（8条）
    daily      日报
    stats      统计信息
    summary    摘要卡片（统计概览）
    sentiment  情感分析
    cat <类型> 分类推送 (stock/forex/commodity/macro/trade/tech/china/us/crypto/energy)
    test       连接测试
    welcome    欢迎卡片

  {BOLD}选项:{RESET}
    --no-cache    禁用缓存，强制重新抓取
    --cache-only  仅使用缓存

  {BOLD}示例:{RESET}
    python news_hub.py fetch
    python news_hub.py feishu compact
    python news_hub.py feishu digest --no-cache
    python news_hub.py feishu alert
    python news_hub.py export csv --limit 20
    python news_hub.py server
    python news_hub.py report
    python news_hub.py db
""")
    else:
        print(f"{YELLOW}Unknown: {cmd}{RESET}")
        print(f"Run: {GREEN}python news_hub.py help{RESET}")


if __name__ == "__main__":
    main()
