def cmd_preview():
    banner("预览", "👁️")
    all_news = fetch_all(quiet=True)
    history = load_history()
    new_news = [n for n in all_news if n['title'] not in history]
    sources = {}
    for n in all_news:
        s = n.get('source', 'Other')
        sources[s] = sources.get(s, 0) + 1
    stats_box(len(all_news), len(new_news), sources)
    info(f"新新闻: {len(new_news)} 条")
    print()
    print(C("  ┌─ 最新消息 ──────────────────────────────────┐"))
    for i, n in enumerate(new_news[:10], 1):
        title = n['title'][:40] + "..." if len(n['title']) > 40 else n['title']
        s = n.get('source', '')
        icon = "🔹" if "金十" in s else ("🔸" if "Investing" in s else "🔶")
        print(C("  │") + f"  {i:>2}. {icon} {title}")
    print(C("  └" + "─" * 46 + "┘"))
    print()
    return 1
