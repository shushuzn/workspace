def fetch_all(quiet=False, use_cache=True, parallel=True, dedup=True):
    config = load_config()
    
    # 检查缓存
    if use_cache:
        cached = get_cached(quiet)
        if cached:
            return cached
    
    start = time.time()
    all_news = []
    sources = {}
    
    if not quiet:
        banner("NewsHub Fetcher")
        info(f"Time: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        print()
    
    # 构建任务列表
    tasks = []
    for key, src in config.get('news_sources', {}).items():
        if src.get('enable', True):
            tasks.append((key, src))
    
    if parallel and len(tasks) > 1:
        # 并行抓取
        if not quiet:
            info(f"并行抓取 {len(tasks)} 个源...")
        
        with ThreadPoolExecutor(max_workers=min(len(tasks), 5)) as executor:
            futures = {executor.submit(fetch_source, k, s): (k, s) for k, s in tasks}
            
            for future in as_completed(futures):
                name, news = future.result()
                sources[name] = len(news)
                all_news.extend(news)
                if not quiet:
                    if news:
                        ok(f"{name}: +{len(news)}")
                    else:
                        warn(f"{name}: 0")
    else:
        # 串行抓取
        for key, src in tasks:
            name = src.get('name', key)
            if not quiet:
                info(f"Fetching {name}...")
            
            news = fetch_source(key, src)[1]
            sources[name] = len(news)
