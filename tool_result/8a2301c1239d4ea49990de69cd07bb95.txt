#!/usr/bin/env python3
"""
NewsHub - 新闻聚合推送入口
"""
import sys
import subprocess
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8')

BASE_DIR = Path(__file__).parent

# 颜色
CYAN = '\033[36m'
GREEN = '\033[32m'
YELLOW = '\033[33m'
RESET = '\033[0m'
BOLD = '\033[1m'


def header(text):
    print(f"\n{BOLD}{CYAN}{'─' * 50}{RESET}")
    print(f"{BOLD}{CYAN}  {text}{RESET}")
    print(f"{BOLD}{CYAN}{'─' * 50}{RESET}\n")


def run(script, *args):
    cmd = [sys.executable, BASE_DIR / script, *args]
    return subprocess.run(cmd).returncode == 0


def main():
    cmd = sys.argv[1] if len(sys.argv) > 1 else "help"
    
    # 别名展开
    from cli_shortcuts import expand_alias, get_alias
    expanded = expand_alias(sys.argv[1:])
    if expanded != sys.argv[1:]:
        cmd = expanded[0]
        sys.argv = [sys.argv[0]] + expanded
    
    # 解析额外参数
    extra_args = sys.argv[3:] if len(sys.argv) > 3 else []
    
    if cmd == "feishu":
        mode = sys.argv[2] if len(sys.argv) > 2 else "all"
        header("Feishu Push")
        run("news_feishu.py", mode, *extra_args)
    
    elif cmd == "qq":
        mode = sys.argv[2] if len(sys.argv) > 2 else "status"
        header("QQ Push")
        run("news_to_qq.py", mode)
    
    elif cmd == "wechat":
        mode = sys.argv[2] if len(sys.argv) > 2 else "test"
        header("WeChat Push")
        run("news_wechat.py", mode)
    
    elif cmd == "fetch":
        header("Fetch News")
        run("news_fetcher.py", *extra_args)
    
    elif cmd == "test":
        header("Test All")
        print(f"{YELLOW}Testing Feishu...{RESET}\n")
        run("news_feishu.py", "all", *extra_args)
        print(f"\n{YELLOW}Testing QQ...{RESET}\n")
        run("news_to_qq.py", "status")
    
    elif cmd == "quick":
        header("Quick Mode")
        run("news_fetcher.py", "--quick")
    
    elif cmd == "export":
        header("Export News")
        run("news_fetcher.py", *sys.argv[2:])
    
    elif cmd == "server":
        header("RSS Server")
        import subprocess
        subprocess.Popen([sys.executable, Path(__file__).parent / "rss_server.py"])
        print(f"  {GREEN}✓{RESET} Server started")

        header("Generate Report")
        run("news_report.py")

        header("Database")
        run("news_db.py")

        header("Trend Analysis")
        run("news_trends.py", *sys.argv[2:])

        header("Subscriptions")
        run("news_subscriptions.py", *sys.argv[2:])
