sys.stdout.reconfigure(encoding='utf-8')
BASE_DIR = Path(__file__).parent
CONFIG_FILE = BASE_DIR / "sources.json"
CACHE_FILE = BASE_DIR / ".news_cache.json"

# 缓存 TTL: 5分钟
CACHE_TTL = 300

# ============================================================
# 缓存
# ============================================================

def load_cache():
    if CACHE_FILE.exists():
        try:
            with open(CACHE_FILE, 'r', encoding='utf-8') as f:
                return json.load(f)
        except:
            pass
    return {"data": [], "time": 0}

def save_cache(news):
    with open(CACHE_FILE, 'w', encoding='utf-8') as f:
        json.dump({"data": news, "time": time.time()}, f, ensure_ascii=False)

def get_cached(quiet=False):
    cache = load_cache()
    age = time.time() - cache.get('time', 0)
    if age < CACHE_TTL:
        if not quiet:
            print(f"  💾 缓存: {len(cache['data'])} 条 ({age:.0f}s前)")
