def gen_compare_table(results):
    """Generate comparison markdown table"""
    lines = [
        "# Stock Comparison",
        "",
        "| Symbol | Price | Target | Upside | Score | Rating | P/E | ROE | FCF |",
        "|--------|-------|--------|--------|-------|--------|-----|-----|-----|",
    ]
    
    for r in results:
        lines.append(
            f"| {r['symbol']} | ${r['price']:.2f} | ${r['target']:.2f} | "
            f"{r['upside']:+.1f}% | {r['score']} | {r['rating']} | "
            f"{r['pe']:.1f}x | {r['roe']:.1f}% | {r['fcf']:.1f}% |"
        )
    
    return "\n".join(lines)


def gen_summary_card(results):
    """Generate summary card"""
    lines = [
        "+" + "-" * 48 + "+",
        "|{:^48}|".format("Stock PRO Summary"),
        "+" + "=" * 48 + "+",
        "",
    ]
    
    for r in results:
        upside_emoji = "🟢" if r['upside'] > 15 else "🟡" if r['upside'] > 0 else "🔴"
        rating_color = "BUY" if r['score'] >= 65 else "HOLD" if r['score'] >= 50 else "SELL"
        
        lines.append(f"{'+' + '-'*48 + '+'}")
        lines.append(f"|{r['symbol']:^10} {'Price':^8} ${r['price']:>7.2f}   |")
        lines.append(f"|{'':^10} {'Target':^8} ${r['target']:>7.2f}   |")
        lines.append(f"|{'':^10} {'Upside':^8} {r['upside']:>+6.1f}% {upside_emoji}  |")
        lines.append(f"|{'':^10} {'Score':^8} {r['score']:>3}/100 ({rating_color}) |")
        lines.append(f"|{'':^10} {'P/E':^8} {r['pe']:>6.1f}x    |")
    
    lines.append("+" + "=" * 48 + "+")
    return "\n".join(lines)
