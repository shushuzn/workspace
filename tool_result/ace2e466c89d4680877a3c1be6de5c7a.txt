async def fetch_category(page, key, source):
    """抓取单个类别的新闻"""
    try:
        print(f"  -> {source['url']}")
        await page.goto(source["url"], timeout=15000)
        await page.wait_for_load_state("networkidle", timeout=10000)
        
        # 获取页面标题和链接
        content = await page.evaluate("""
            () => {
                const headlines = [];
                
                // 获取页面标题
                const pageTitle = document.title || '';
                
                // 获取所有链接的标题
                const links = document.querySelectorAll('a[title]');
                links.forEach(a => {
                    const text = a.title.trim() || a.textContent.trim();
                    if (text && text.length > 5 && text.length < 150) {
                        headlines.push(text);
                    }
                });
                
                // 获取 h1-h3
                ['h1', 'h2', 'h3'].forEach(tag => {
                    document.querySelectorAll(tag).forEach(el => {
                        const text = el.textContent.trim();
                        if (text && text.length > 5 && text.length < 150) {
                            headlines.push(text);
                        }
                    });
                });
                
                // 去重
                return [...new Set(headlines)].slice(0, 8);
            }
        """)
        
        return key, source["name"], content if content else []
    
    except Exception as e:
        print(f"  [!] Error: {str(e)[:50]}")
        return key, source["name"], []
