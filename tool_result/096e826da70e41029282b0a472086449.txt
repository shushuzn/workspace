"""Cron scheduler for Stock PRO"""
import time
import threading
from datetime import datetime
from pathlib import Path

WORKSPACE = Path("D:/OpenClaw/workspace")
CRON_FILE = WORKSPACE / "30-scripts-tools" / "stock_pro_cron.json"

def parse_cron_next(cron_expr):
    """Parse cron expression and return next run time (simplified)"""
    # Simple parser for "HH:MM" format
    if ":" in cron_expr:
        parts = cron_expr.split(":")
        hour, minute = int(parts[0]), int(parts[1])
        now = datetime.now()
        next_run = now.replace(hour=hour, minute=minute, second=0, microsecond=0)
        if next_run <= now:
            from datetime import timedelta
            next_run += timedelta(days=1)
        return next_run
    return None

class CronJob:
    def __init__(self, name, symbols, command, schedule):
        self.name = name
        self.symbols = symbols
        self.command = command
        self.schedule = schedule
        self.next_run = parse_cron_next(schedule)
        self.active = True
    
    def to_dict(self):
        return {
            "name": self.name,
            "symbols": self.symbols,
            "command": self.command,
            "schedule": self.schedule,
            "next_run": self.next_run.isoformat() if self.next_run else None,
            "active": self.active
        }

class CronScheduler:
    def __init__(self):
        self.jobs = []
        self.running = False
        self.thread = None
        self.load()
    
    def load(self):
        import json
        if CRON_FILE.exists():
            try:
                with open(CRON_FILE, 'r') as f:
                    data = json.load(f)
                    for j in data.get("jobs", []):
                        job = CronJob(j["name"], j["symbols"], j["command"], j["schedule"])
                        job.active = j.get("active", True)
                        self.jobs.append(job)
            except:
                pass
    
    def save(self):
        import json
        data = {"jobs": [j.to_dict() for j in self.jobs]}
        with open(CRON_FILE, 'w') as f:
            json.dump(data, f, indent=2)
    
    def add(self, name, symbols, command, schedule):
        job = CronJob(name, symbols, command, schedule)
        self.jobs.append(job)
        self.save()
        return f"[Cron] Added job '{name}' - schedule: {schedule}"
    
    def remove(self, name):
        self.jobs = [j for j in self.jobs if j.name != name]
        self.save()
        return f"[Cron] Removed job '{name}'"
    
    def list_jobs(self):
        if not self.jobs:
            return "[Cron] No jobs scheduled. Add with: --cron-add \"09:00\" NVDA META"
        
        lines = ["[Cron] Scheduled Jobs:", "-" * 50]
        for j in self.jobs:
            status = "ACTIVE" if j.active else "PAUSED"
            next_time = j.next_run.strftime("%Y-%m-%d %H:%M") if j.next_run else "N/A"
            lines.append(f"  [{status}] {j.name} - {j.command} {j.symbols} @ {j.schedule}")
            lines.append(f"           Next: {next_time}")
        return "\n".join(lines)
    
    def pause(self, name):
        for j in self.jobs:
            if j.name == name:
                j.active = False
                self.save()
                return f"[Cron] Paused job '{name}'"
        return f"[Cron] Job '{name}' not found"
    
    def resume(self, name):
        for j in self.jobs:
            if j.name == name:
                j.active = True
                self.save()
                return f"[Cron] Resumed job '{name}'"
        return f"[Cron] Job '{name}' not found"
    
    def execute_job(self, job):
        """Execute a cron job"""
        from .core import analyze_multiple
        symbols = job.symbols
        command = job.command
        
        results = analyze_multiple(symbols)
        
        if command == "alert":
            from .integrations import check_alerts
            return check_alerts(results)
        elif command == "report":
            from .reports import gen_report
            for r in results:
                gen_report(r)
            return f"[Cron] Generated reports for {symbols}"
        elif command == "screener":
            from .screener import StockScreener
            return StockScreener().show()
        else:
            return f"[Cron] Unknown command: {command}"
    
    def run(self):
        """Background runner"""
        while self.running:
            now = datetime.now()
            for job in self.jobs:
                if job.active and job.next_run and now >= job.next_run:
                    print(f"[Cron] Executing: {job.name}")
                    result = self.execute_job(job)
                    print(f"[Cron] Result: {result}")
                    job.next_run = parse_cron_next(job.schedule)
            time.sleep(60)
    
    def start(self):
        if not self.running:
            self.running = True
            self.thread = threading.Thread(target=self.run, daemon=True)
            self.thread.start()
            return "[Cron] Scheduler started"
        return "[Cron] Scheduler already running"
    
    def stop(self):
        self.running = False
        return "[Cron] Scheduler stopped"
