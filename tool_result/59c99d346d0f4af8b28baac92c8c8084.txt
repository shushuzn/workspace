def print_summary_card(results):
    """Print summary card for stock"""
    r = results
    upside = r['upside']
    score = r['score']
    
    # Rating indicator
    if upside > 30:
        rating_icon = "[BUY ]"
    elif upside > 15:
        rating_icon = "[BUY ]"
    elif upside > 0:
        rating_icon = "[HOLD]"
    else:
        rating_icon = "[SELL]"
    
    # Score bar
    score_bar = "=" * int(score/5) + "-" * (20 - int(score/5))
    
    price_str = f"${r['price']:>8.2f}"
    target_str = f"${r['target']:>8.2f}"
    upside_str = f"{upside:>+6.1f}%"
    
    print(f"""
+-------------------------------------------------------------+
| {r['symbol']:<6} {price_str}  ->  {target_str}  {rating_icon} {upside_str}
+-------------------------------------------------------------+
| Score: [{score_bar}] {score}/100
| P/E: {r['pe']:>5.1f}x  |  PEG: {r['peg']:>4.2f}  |  FCF: {r['fcf_yield']*100:>5.1f}%
| ROE: {r['roe']*100:>5.1f}%  |  Div: {r['div_yield']*100:>5.2f}%  |  Beta: {r['beta']:>4.2f}
+-------------------------------------------------------------+
    """)
