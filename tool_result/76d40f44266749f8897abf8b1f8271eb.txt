def cmd_digest(limit=6):

    if not config.get("app_id"):
        error("未配置飞书应用")
        return

    from core.async_fetcher import fetch_all
    from analyze.quality_scorer import rank_news_by_quality

    info("抓取新闻...")
    news = fetch_all(quiet=True)
    info(f"获取 {len(news)} 条新闻")

    if not news:
        warn("无新闻数据")
        return

    # 按质量评分排序，选取高质量新闻
    ranked = rank_news_by_quality(news)
    news = ranked[:limit]
