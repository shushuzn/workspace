def score_sentiment_intensity(score: int) -> float:
    """情感强度评分 (0-0.5)"""
    # 情感越强(正向或负向)越有价值
    return min(0.5, abs(score) / 100)


def calculate_quality_score(news: Dict) -> float:
    """计算综合质量评分 (0-100)"""
    weights = {
        "source": 0.25,
        "keyword": 0.20,
        "recency": 0.20,
        "data": 0.15,
        "title_length": 0.10,
        "sentiment": 0.10,
    }

    title = news.get("title", "")

    scores = {
        "source": score_source_weight(news.get("source", "")),
        "keyword": score_keyword_match(title),
        "recency": score_recency(news.get("time", "")),
        "data": score_has_data(title),
        "title_length": score_title_length(title),
        "sentiment": score_sentiment_intensity(news.get("score", 0)),
    }

    # 计算加权总分
    total = sum(scores[k] * weights[k] for k in weights)

    return round(total * 100, 1)


def rank_news_by_quality(news_list: List[Dict]) -> List[Dict]:

    return sorted(news_list, key=lambda x: x.get("quality_score", 0), reverse=True)
