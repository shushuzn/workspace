def analyze(title):
    pos = sum(1 for k in POSITIVE if k in title)
    neg = sum(1 for k in NEGATIVE if k in title)
    sentiment = "利好" if pos > neg else ("利空" if neg > pos else "中性")
    sector = "通用"
    for sec, kws in SECTORS.items():
        if any(k in title for k in kws):
            sector = sec
            break
    codes = re.findall(r'[0-9]{6}', title)
    codes.extend(re.findall(r'\b[A-Z]{2,5}\b', title))
    codes = list(dict.fromkeys(codes))[:2]
    keywords = []
    for sec, kws in SECTORS.items():
        for kw in kws:
            if kw in title:
                keywords.append(kw)
                break
    return sentiment, sector, keywords[:3], codes

def get_summary(title):
    for p in ["华为","阿里","腾讯","小米","比亚迪","特斯拉","茅台","OpenAI","英伟达","苹果","微软"]:
        if p in title: return p
    if any(k in title for k in ["上涨","大涨"]): return "上涨"
    if any(k in title for k in ["下跌","大跌"]): return "下跌"
    if "突破" in title: return "突破"
    if any(k in title for k in ["风险","警告"]): return "警示"
    return ""

def get_signal(news):
    title, sent = news.get("title",""), news.get("sentiment","")
    if sent == "利好":
        if any(k in title for k in ["涨停","暴涨"]): return "强势买入"
        if any(k in title for k in ["突破","创新高"]): return "关注买入"
    elif sent == "利空":
        if any(k in title for k in ["跌停","暴跌"]): return "坚决回避"
        if any(k in title for k in ["亏损","造假"]): return "风险警示"
    return ""

def get_risk(news_list):
    score = sum(2 if n.get("sentiment")=="利空" else 0 for n in news_list)
    return "高风险" if score >= 8 else ("中风险" if score >= 4 else "低风险")
