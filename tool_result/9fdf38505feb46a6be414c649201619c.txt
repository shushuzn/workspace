def get_rss_sources() -> List[tuple]:
    config = load_config()
    sources = []
    
    # 读取内置源
    for key, src in config.get("news_sources", {}).items():
        if src.get("type") == "rss" and src.get("enable", True):
            sources.append((src["name"], src.get("url", ""), src.get("limit", 20)))
    
    # 读取自定义源
    custom_config = config.get("custom_rss", {})
    if custom_config.get("enabled", True):
        for src in custom_config.get("sources", []):
            if src.get("enable", True) and src.get("url"):
                sources.append((src.get("name", "自定义"), src["url"], src.get("limit", 15)))
    
    return sources
