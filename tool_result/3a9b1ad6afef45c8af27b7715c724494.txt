# 获取外盘行情（模拟数据，实际可接API）
def get_external_markets() -> str:
    """获取外盘行情摘要"""
    try:
        # 尝试获取真实数据
        url = "https://hq.sinajs.cn/list=gb_ixic,gb_ipsa,gb_inx"
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=5) as resp:
            data = resp.read().decode('gbk')
        
        markets = []
        for line in data.strip().split('\n'):
            if '=' in line:
                symbol = line.split('=')[0].split('_')[-1]
                values = line.split('"')[1].split(',')
                if len(values) >= 4:
                    name = {"ixic": "纳指", "ipsa": "标普", "inx": "道指"}.get(symbol, symbol)
                    price = float(values[0])
                    change = float(values[1])
                    pct = float(values[2])
                    direction = "+" if change >= 0 else ""
                    markets.append(f"{name}:{direction}{change:.0f}({direction}{pct:.2f}%)")
        
        return " | ".join(markets[:3])
    except:
        return "外盘: 获取失败"
