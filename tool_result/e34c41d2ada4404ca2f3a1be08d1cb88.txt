def get_fig_color(score):
    """Get color based on score"""
    if score >= 75: return COLORS['buy']
    elif score >= 60: return '#7bed9f'
    elif score >= 40: return COLORS['hold']
    else: return COLORS['sell']

def create_stock_card(data):
    """Create stock info card"""
    if not data:
        return html.Div("No data", style=CARD_STYLE)
    
    score_color = get_fig_color(data['score'])
    
    return html.Div([
        html.H3(f"{data['symbol']} - {data.get('name', '')}", 
                style={'color': COLORS['text'], 'marginBottom': '10px'}),
        html.Div([
            html.Div([
                html.Span("Score: ", style={'fontWeight': 'bold'}),
                html.Span(f"{data['score']}", style={'color': score_color, 'fontSize': '24px'})
            ], style={'display': 'inline-block', 'marginRight': '20px'}),
            html.Div([
                html.Span("Upside: ", style={'fontWeight': 'bold'}),
                html.Span(f"{data['upside']:.1f}%", 
                         style={'color': COLORS['buy'] if data['upside'] > 0 else COLORS['sell']})
            ], style={'display': 'inline-block'}),
        ]),
        html.Hr(style={'borderColor': COLORS['accent']}),
        html.Table([
            html.Tr([html.Td("Price"), html.Td(f"${data['price']:.2f}")]),
            html.Tr([html.Td("Target"), html.Td(f"${data['target']:.2f}")]),
            html.Tr([html.Td("P/E"), html.Td(f"{data['pe']:.1f}x")]),
            html.Tr([html.Td("ROE"), html.Td(f"{data['roe']:.1f}%")]),
            html.Tr([html.Td("Div"), html.Td(f"{data['div']:.2f}%")]),
        ], style={'width': '100%', 'fontSize': '14px'}),
    ], style=CARD_STYLE)

def create_bar_chart(stocks, metric='score'):
    """Create bar chart"""
    if not stocks:
        return go.Figure()
    
    symbols = [s['symbol'] for s in stocks]
    values = [s.get(metric, 0) for s in stocks]
    colors = [get_fig_color(s.get('score', 50)) for s in stocks]
    
    fig = go.Figure(data=[
        go.Bar(x=symbols, y=values, marker_color=colors)
    ])
    
    fig.update_layout(
        paper_bgcolor=COLORS['card_bg'],
        plot_bgcolor=COLORS['card_bg'],
        font_color=COLORS['text'],
        title=dict(text=metric.upper(), font=dict(size=16)),
        margin=dict(l=20, r=20, t=40, b=20),
        xaxis=dict(gridcolor='#2a2a4a'),
        yaxis=dict(gridcolor='#2a2a4a'),
    )
    
    return fig

# Callbacks
@app.callback(
    Output('stock-card', 'children'),
    Input('analyze-btn', 'n_clicks'),
    State('stock-input', 'value')
)
def update_stock_card(n_clicks, symbol):
    if not symbol or n_clicks == 0:
        return html.Div("Enter a stock symbol and click Analyze", 
                       style={'textAlign': 'center', **CARD_STYLE})
    
    try:
        data = analyze(symbol.upper())
        if data:
            return create_stock_card(data)
        return html.Div(f"Stock {symbol} not found", style=CARD_STYLE)
    except Exception as e:
        return html.Div(f"Error: {str(e)}", style=CARD_STYLE)

@app.callback(
    [Output('top-picks-chart', 'children'),
     Output('top-picks-table', 'children')],
    Input('top-picks-container', 'id')
)
def update_top_picks(_):
    try:
        picks = top_picks(10)
        fig = create_bar_chart(picks, 'score')
        cards = [create_stock_card(p) for p in picks[:5]]
        return dcc.Graph(figure=fig), cards
    except Exception as e:
        return html.Div(f"Error: {str(e)}"), []
