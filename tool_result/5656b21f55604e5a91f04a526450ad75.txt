def get_cached(quiet=False):
    cache = load_cache()
    age = time.time() - cache.get('time', 0)
    if age < CACHE_TTL:
        if not quiet:
            print(f"  💾 缓存: {len(cache['data'])} 条 ({time_ago(cache.get('time', 0))})")
        return cache.get('data', [])
    return None

def summary(total, sources, elapsed, dup_removed=0):
    print()
    print("─" * 50)
    print(f"  📊 Total: {total}", end="")
    if dup_removed > 0:
        print(f" (去重: -{dup_removed})", end="")
    print(f"  ⏱️ {elapsed:.1f}s")
    for name, count in sources.items():
        print(f"     {name}: {count}")
    print("─" * 50)
    print()

def get_ssl():
    ctx = ssl.create_default_context()
    ctx.check_hostname = False
    ctx.verify_mode = ssl.CERT_NONE
    return ctx
