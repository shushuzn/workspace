def filter_by_keywords(news_list, keywords):
    """按关键词筛选（AND/OR）"""
    if not keywords:
        return news_list
    result = []
    for n in news_list:
        title = n.get('title', '')
        if any(kw in title for kw in keywords):
            result.append(n)
    return result

def sort_by_relevance(news_list, keywords):
    """按关键词相关性排序"""
    if not keywords:
        return news_list
    scored = []
    for n in news_list:
        title = n.get('title', '')
        score = sum(1 for kw in keywords if kw in title)
        n['relevance'] = score
        scored.append(n)
    return sorted(scored, key=lambda x: x.get('relevance', 0), reverse=True)
