    data = {"app_id": cfg['app_id'], "app_secret": cfg['app_secret']}

def send_card(token, receive_id, card, retry=2):
    """发送卡片，支持重试"""
    cfg = load_feishu_config()
    api_url = "https://open.feishu.cn/open-apis/im/v1/messages"
    params = f"?receive_id_type={cfg.get('receive_id_type', 'open_id')}"
    
    payload = {
        "receive_id": receive_id,
        "msg_type": "interactive",
        "content": json.dumps(card, ensure_ascii=False)
    }
    
    req = urllib.request.Request(
        api_url + params,
        data=json.dumps(payload, ensure_ascii=False).encode('utf-8'),
        headers={
            'Content-Type': 'application/json',
            'Authorization': f'Bearer {token}'
        },
        method='POST'
    )
    
    ctx = ssl.create_default_context()
    for attempt in range(retry):
        try:
            with urllib.request.urlopen(req, timeout=30, context=ctx) as r:
                return json.loads(r.read())
        except Exception as e:
            if attempt < retry - 1:
                time.sleep(1)
            else:
                return {"code": -1, "msg": str(e)}
