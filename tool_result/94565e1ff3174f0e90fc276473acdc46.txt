def news_to_html(news_list):
    """将新闻列表转换为HTML"""
    html = f"""<!DOCTYPE html>
<html lang="zh">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>NewsHub - {len(news_list)} 条新闻</title>
    <style>
        body {{ font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif; max-width: 800px; margin: 0 auto; padding: 20px; background: #1a1a2e; color: #eee; }}
        h1 {{ color: #00d4ff; border-bottom: 2px solid #00d4ff; padding-bottom: 10px; }}
        .meta {{ color: #888; font-size: 0.9em; margin-bottom: 20px; }}
        .news {{ background: #16213e; border-radius: 8px; padding: 15px; margin: 10px 0; border-left: 4px solid #00d4ff; }}
        .news h3 {{ margin: 0 0 8px 0; color: #fff; }}
        .news a {{ color: #00d4ff; text-decoration: none; }}
        .news a:hover {{ text-decoration: underline; }}
        .news .source {{ color: #888; font-size: 0.85em; margin-top: 5px; }}
        .tag {{ display: inline-block; background: #0f3460; padding: 2px 8px; border-radius: 4px; font-size: 0.8em; margin-right: 5px; }}
        .rss-link {{ background: #ff6b35; color: white; padding: 10px 20px; border-radius: 5px; text-decoration: none; display: inline-block; margin-bottom: 20px; }}
    </style>
</head>
<body>
    <h1>📊 NewsHub</h1>
    <div class="meta">共 {len(news_list)} 条新闻 | 更新时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}</div>
    <a class="rss-link" href="/news.xml">📡 RSS订阅</a>
"""
    
    for n in news_list[:30]:
        source = n.get('source', '')
        source_icon = "📰" if "新浪" in source else ("🌐" if "Investing" in source else ("📊" if "Seeking" in source else "📱"))
        html += f"""
    <div class="news">
        <h3>{source_icon} {n.get('title', '')}</h3>
        <a href="{n.get('url', '')}" target="_blank">🔗 查看详情</a>
        <div class="source">来源: {source}</div>
    </div>
"""
    
    html += """
</body>
</html>"""
