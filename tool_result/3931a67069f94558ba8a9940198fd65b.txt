def enrich_news(news_list: List[Dict]) -> List[Dict]:
    """增强新闻（添加分类、重要性、情感）"""
    enriched = []

    for news in news_list:
        title = news.get("title", "")

        item = dict(news)

        if not item.get("category"):
            item["category"] = classify_news(title)

        if not item.get("importance"):
            item["importance"] = get_importance(title)

        if item.get("score", None) is None:
            item["score"] = get_sentiment_score(title)

        if not item.get("hash"):
            item["hash"] = compute_hash(title)

        enriched.append(item)

    return enriched
