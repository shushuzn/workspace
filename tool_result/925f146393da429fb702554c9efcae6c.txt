# Cache system
class Cache:
    def __init__(self):
        self.data = {}
        self.lock = threading.Lock()
    
    def get(self, key, ttl=300):  # 5 minutes default
        with self.lock:
            if key in self.data:
                data, timestamp = self.data[key]
                if datetime.now() - timestamp < timedelta(seconds=ttl):
                    return data
                else:
                    del self.data[key]
            return None
    
    def set(self, key, value):
        with self.lock:
            self.data[key] = (value, datetime.now())
    
    def clear(self):
        with self.lock:
            self.data.clear()

cache = Cache()
