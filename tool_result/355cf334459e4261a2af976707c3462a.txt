def cmd_add(keyword, level="medium", desc=""):
    """添加订阅"""
    if add_subscription(keyword, level, desc):
        print(f"✓ 已添加: {keyword} ({level})")
    else:
        print(f"✗ 已存在: {keyword}")

    if remove_subscription(keyword):

        print(f"✗ 未找到: {keyword}")
