def _fetch(url):
    if url in _cache:
        ts, data = _cache[url]
        if time.time() - ts < _cache_ttl:
            return data
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=15) as resp:
            data = resp.read()
            _cache[url] = (time.time(), data)
            return data
    except:
        return None
