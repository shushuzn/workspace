    def reset(): return Colors.RESET

    """带颜色输出"""
    return f"{color}{text}{Colors.RESET}"

    """利好(绿)"""
    return f"{Colors.pos()}{text}{Colors.RESET}"

def cn(text):
    """利空(红)"""
    return f"{Colors.neg()}{text}{Colors.RESET}"

def cne(text):
    """中性(灰)"""
    return f"{Colors.neu()}{text}{Colors.RESET}"
