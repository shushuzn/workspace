def fetch(sym):

def calc_dcf(q, fin):
    price, nm, roic, beta = q["price"], fin["nm"], fin["roic"], q["beta"]
    wacc = 0.08 + beta * 0.04
    g_term = min(roic * 0.5, 0.035)
    pv = 0
    for y in range(1, 6):
        g = min(g_term * y, 0.20)
        fcf = price * nm * 0.6 * (1 + g)
        pv += fcf / (1 + wacc) ** y
    tv = (price * nm * 0.6 * 1.04) / (wacc - g_term)
    pv += tv / (1 + wacc) ** 5
    return pv
