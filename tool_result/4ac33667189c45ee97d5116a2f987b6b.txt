def filter_by_importance(news_list, level):
    """按重要性筛选"""
    return [n for n in news_list if n.get('importance_level') == level]

def sort_by_importance(news_list):
    """按重要性排序"""
    return sorted(news_list, key=lambda x: x.get('importance', 0), reverse=True)
