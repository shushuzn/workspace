NewsHub Analyzer - 分析功能
"""
import re
import hashlib
import time
import json
from concurrent.futures import ThreadPoolExecutor

OLLAMA_URL = "http://localhost:11434/api/generate"
OLLAMA_MODEL = "qwen2.5:1.5b"
LLM_CACHE_FILE = ".llm_cache.json"
LLM_CACHE_TTL = 3600

def load_llm_cache():
    try:
        with open(LLM_CACHE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except:
        return {}

def save_llm_cache(cache):
    with open(LLM_CACHE_FILE, "w", encoding="utf-8") as f:
        json.dump(cache, f, ensure_ascii=False)

def check_ollama_status():
    import ssl, urllib.request
    try:
        req = urllib.request.Request("http://localhost:11434/api/tags", headers={"Content-Type": "application/json"})
        with urllib.request.urlopen(req, timeout=5, context=ssl.create_default_context()) as resp:
            return json.loads(resp.read().decode("utf-8")).get("models", []) != []
    except:
        return False

def call_ollama(prompt):
    import ssl, urllib.request
    try:
        data = {
            "model": OLLAMA_MODEL,
            "prompt": prompt,
            "stream": False,
            "options": {"num_predict": 200, "temperature": 0.3}
        }
        req = urllib.request.Request(OLLAMA_URL, data=json.dumps(data).encode("utf-8"), headers={"Content-Type": "application/json"}, method="POST")
        with urllib.request.urlopen(req, timeout=30, context=ssl.create_default_context()) as resp:
            return json.loads(resp.read().decode("utf-8")).get("response", "")
    except Exception as e:
        return ""

def analyze_with_llm(title):
    title_hash = hashlib.md5(title.encode()).hexdigest()
    cache = load_llm_cache()
    if title_hash in cache:
        return cache[title_hash]
    if not check_ollama_status():
        return None
    prompt = f"""分析新闻: {title}
返回JSON: {{"sentiment":"positive|negative|neutral","sentiment_score":-100到100,"urgency":"high|medium|low","markets":["stock"],"entities":[],"keywords":[],"summary":"15字摘要","advice":"bullish|bearish|neutral"}}"""
    response = call_ollama(prompt)
    if response:
        try:
            import json as j
            for line in response.split("\n"):
                line = line.strip()
                if line.startswith("{") and line.endswith("}"):
                    result = j.loads(line)
                    cache[title_hash] = result
                    save_llm_cache(cache)
                    return result
        except:
            pass
    return None
