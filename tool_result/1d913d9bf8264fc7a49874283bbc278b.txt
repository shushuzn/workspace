var G = null;

function init() {
    initTheme();
    G = loadGame();
    renderBuildings();
    renderUpgrades();
    renderAchievements();
    renderStats();
    updateUI();
    updateStats();
    initTabs();
    startGameLoop();
    startAutoSave(G);
    document.getElementById('gold-click-area').onclick = clickGold;
    document.getElementById('save-btn').onclick = function() { saveGame(G); showMsg('已保存!', 'success'); };
    document.getElementById('reset-btn').onclick = function() { if(confirm('重置游戏?')) location.reload(); };
    document.getElementById('theme-btn').onclick = toggleTheme;
}

function clickGold(e) {
    var v = (G.upgrades && G.upgrades.indexOf('click_power') !== -1) ? 2 : 1;
    G.gold += v;
    G.totalEarned += v;
    G.totalClicks = (G.totalClicks || 0) + 1;
    var el = document.createElement('div');
    el.className = 'floating-number';
    el.textContent = '+' + v;
    el.style.left = (e.clientX - 20) + 'px';
    el.style.top = (e.clientY - 20) + 'px';
    document.body.appendChild(el);
    setTimeout(function() { if(el.parentNode) el.parentNode.removeChild(el); }, 1000);
    updateUI();
    checkAchievements();
}

function initTabs() {
    var tabs = document.querySelectorAll('.tab');
    for (var i = 0; i < tabs.length; i++) {
        tabs[i].onclick = function() {
            var allTabs = document.querySelectorAll('.tab');
            var allContent = document.querySelectorAll('.tab-content');
            for (var j = 0; j < allTabs.length; j++) allTabs[j].classList.remove('active');
            for (var k = 0; k < allContent.length; k++) allContent[k].classList.remove('active');
            this.classList.add('active');
            document.getElementById('tab-' + this.getAttribute('data-tab')).classList.add('active');
        };
    }
}

function renderBuildings() {
    var c = document.getElementById('building-list');
    if (!c || !G) return;
    c.innerHTML = '';
    for (var i = 0; i < BUILDINGS.length; i++) {
        var b = BUILDINGS[i];
        var owned = G.buildings[b.id] || 0;
        var cost = Math.floor(b.baseCost * Math.pow(1.15, owned));
        var prod = b.baseProduction * owned * getBuildingMultiplier(G, b.id);
        var unlocked = !b.unlockAt || G.totalEarned >= b.unlockAt;
        var affordable = G.gold >= cost;
        var div = document.createElement('div');
        div.className = 'building' + (!unlocked ? ' locked' : '') + (owned > 0 ? ' owned' : '') + (affordable ? ' affordable' : '');
        div.innerHTML = '<span class="building-icon" style="color:' + b.color + '">' + b.icon + '</span>' +
            '<div class="building-info"><div class="building-name">' + b.icon + ' ' + b.name + '</div><div class="building-desc">' + b.description + '</div></div>' +
            '<div class="building-production">+' + formatNumber(prod) + '/s</div>' +
            '<div class="building-count">' + owned + '</div>' +
            '<div class="building-cost">' + formatNumber(cost) + '</div>';
        if (unlocked) div.onclick = (function(bb) { return function() { buyBuilding(bb); }; })(b);
        c.appendChild(div);
    }
}

function buyBuilding(b) {
    var owned = G.buildings[b.id] || 0;
    var cost = Math.floor(b.baseCost * Math.pow(1.15, owned));
    if (G.gold >= cost) {
        G.gold -= cost;
        G.buildings[b.id] = owned + 1;
        renderBuildings();
        renderStats();
        updateUI();
        updateStats();
    }
}

function renderUpgrades() {
    var c = document.getElementById('upgrade-list');
    if (!c || !G) return;
    c.innerHTML = '';
    var purchased = G.upgrades || [];
    var available = UPGRADES.filter(function(u) { return purchased.indexOf(u.id) === -1; });
    if (available.length === 0) { c.innerHTML = '<p class="empty">所有升级已购买!</p>'; return; }
    for (var i = 0; i < available.length; i++) {
        var u = available[i];
        var affordable = G.gold >= u.cost;
        var div = document.createElement('div');
        div.className = 'upgrade' + (affordable ? ' affordable' : '');
        div.innerHTML = '<span class="upgrade-icon">' + u.icon + '</span>' +
            '<div class="upgrade-info"><div class="upgrade-name">' + u.name + '</div><div class="upgrade-desc">' + u.desc + '</div></div>' +
            '<div class="upgrade-cost">' + formatNumber(u.cost) + '</div>';
        div.onclick = (function(uu) { return function() { buyUpgrade(uu); }; })(u);
        c.appendChild(div);
    }
}

function buyUpgrade(u) {
    if (G.gold >= u.cost && (G.upgrades || []).indexOf(u.id) === -1) {
        G.gold -= u.cost;
        if (!G.upgrades) G.upgrades = [];
        G.upgrades.push(u.id);
        showMsg('升级: ' + u.name, 'success');
        renderUpgrades();
        renderBuildings();
        updateUI();
    }
}

function renderAchievements() {
    var c = document.getElementById('achievement-list');
    if (!c || !G) return;
    c.innerHTML = '';
    var unlocked = G.achievements || [];
    for (var i = 0; i < ACHIEVEMENTS.length; i++) {
        var a = ACHIEVEMENTS[i];
        var div = document.createElement('div');
        div.className = 'achievement' + (unlocked.indexOf(a.id) !== -1 ? ' unlocked' : ' locked');
        div.innerHTML = '<span class="achievement-icon">' + a.icon + '</span>' +
            '<div class="achievement-info"><div class="achievement-name">' + a.name + '</div><div class="achievement-desc">' + a.desc + '</div></div>';
        c.appendChild(div);
    }
}

function renderStats() {
    var c = document.getElementById('stats-detail');
    if (!c || !G) return;
    var gps = getTotalProduction(G);
    var total = getTotalBuildings(G);
    var time = Math.floor((Date.now() - G.startTime) / 1000);
    var achs = (G.achievements || []).length;
    c.innerHTML = '<div class="stat-card"><div class="label">金币</div><div class="value">' + formatNumber(G.gold) + '</div></div>' +
        '<div class="stat-card"><div class="label">产量/s</div><div class="value">' + formatNumber(gps) + '</div></div>' +
        '<div class="stat-card"><div class="label">累计</div><div class="value">' + formatNumber(G.totalEarned) + '</div></div>' +
        '<div class="stat-card"><div class="label">时间</div><div class="value">' + formatTime(time) + '</div></div>' +
        '<div class="stat-card"><div class="label">建筑</div><div class="value">' + total + '</div></div>' +
        '<div class="stat-card"><div class="label">成就</div><div class="value">' + achs + '/' + ACHIEVEMENTS.length + '</div></div>';
}

function updateUI() {
    if (!G) return;
    document.getElementById('gold').textContent = formatNumber(G.gold);
    document.getElementById('gps').textContent = formatNumber(getTotalProduction(G));
}

function updateStats() {
    if (!G) return;
    document.getElementById('stat-buildings').textContent = getTotalBuildings(G);
    document.getElementById('stat-total').textContent = formatNumber(G.totalEarned);
    document.getElementById('stat-time').textContent = formatTime(Math.floor((Date.now() - G.startTime) / 1000));
    document.getElementById('stat-achievements').textContent = (G.achievements || []).length + '/' + ACHIEVEMENTS.length;
}

var lastTick = Date.now();
function gameLoop() {
    var now = Date.now();
    var delta = (now - lastTick) / 1000;
    lastTick = now;
    if (!G) return;
    var gps = getTotalProduction(G);
    G.gold += gps * delta;
    G.totalEarned += gps * delta;
    updateUI();
    if (Math.random() < 0.02) { renderBuildings(); renderStats(); }
    checkAchievements();
}

function startGameLoop() {
    setInterval(gameLoop, 100);
}

document.addEventListener('DOMContentLoaded', init);
