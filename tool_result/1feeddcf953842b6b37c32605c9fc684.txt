"""Dashboard - Real-time monitoring dashboard"""
import json
from pathlib import Path
from datetime import datetime
from stock_pro.core import A, analyze_multiple
from stock_pro.sectors import get_all_sectors, get_symbols_by_sector

WORKSPACE = Path("D:/OpenClaw/workspace")
DASHBOARD_FILE = WORKSPACE / "30-scripts-tools" / "stock_pro_dashboard.json"


class Dashboard:
    """Real-time monitoring dashboard"""
    
    def __init__(self):
        self.data = self._load()
    
    def _load(self):
        """Load dashboard data"""
        if DASHBOARD_FILE.exists():
            try:
                with open(DASHBOARD_FILE, 'r') as f:
                    return json.load(f)
            except:
                pass
        return {"last_update": None, "portfolio": {}, "sectors": []}
    
    def _save(self):
        """Save dashboard data"""
        self.data["last_update"] = datetime.now().isoformat()
        with open(DASHBOARD_FILE, 'w') as f:
            json.dump(self.data, f, indent=2)
    
    def update_all(self):
        """Update all dashboard data"""
        from stock_pro.core import A
        symbols = list(A.keys())
        results = analyze_multiple(symbols)
        
        # Portfolio overview
        total_value = sum(r["price"] for r in results)
        total_target = sum(r["target"] for r in results)
        
        self.data["portfolio"] = {
            "stocks": len(results),
            "total_value": total_value,
            "total_target": total_target,
            "avg_upside": (total_target - total_value) / total_value * 100,
            "avg_score": sum(r["score"] for r in results) / len(results)
        }
        
        # Top movers
        sorted_upside = sorted(results, key=lambda x: x["upside"], reverse=True)
        self.data["top_gainers"] = [{"symbol": r["symbol"], "upside": r["upside"]} for r in sorted_upside[:5]]
        
        sorted_score = sorted(results, key=lambda x: x["score"], reverse=True)
        self.data["top_scores"] = [{"symbol": r["symbol"], "score": r["score"]} for r in sorted_score[:5]]
        
        # Sector summary
        sectors = get_all_sectors()
        sector_data = []
        for sector in sectors:
            syms = get_symbols_by_sector(sector)
            sector_results = [r for r in results if r["symbol"] in syms]
            if sector_results:
                sector_data.append({
                    "name": sector,
                    "stocks": len(sector_results),
                    "avg_score": sum(r["score"] for r in sector_results) / len(sector_results),
                    "avg_upside": sum(r["upside"] for r in sector_results) / len(sector_results)
                })
        
        sector_data.sort(key=lambda x: x["avg_score"], reverse=True)
        self.data["sectors"] = sector_data
        
        self._save()
        return self.data
