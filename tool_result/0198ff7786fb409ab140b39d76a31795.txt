def fetch_rss(url, name, limit=10):
    news = []
    try:
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=10, context=get_ssl_context()) as resp:
            root = ET.fromstring(resp.read().decode('utf-8'))
            for item in root.findall('.//item')[:limit]:
                title = item.findtext('title', '').strip()
                link = item.findtext('link', '')
                if title and link:
                    n = {'title': title, 'url': link, 'source': name, 'ts': time.time()}
                    n['category'] = classify_news(title)
                    n['sentiment'], n['score'] = analyze_sentiment(title)
                    news.append(n)
    except:
        pass
    return news

def fetch_all(quiet=False):
    cache = load_cache()
    if cache.get('news') and time.time() - cache.get('time', 0) < CACHE_TTL:
        if not quiet:
            elapsed = int((time.time() - cache.get('time', 0)) / 60)
            print(f'  💾 缓存: {len(cache["news"])} 条 ({elapsed}m前)')
        return cache['news']
    
    results = []
    start = time.time()
    if not quiet:
        print('  ▶ 并行抓取...')
    
    with ThreadPoolExecutor(max_workers=5) as executor:
        futures = {executor.submit(src[1] if callable(src[1]) else src['func']): src[0] if isinstance(src, tuple) else src['name'] for src in SOURCES}
        for future in futures:
            name = futures[future]
            try:
                news = future.result()
                results.extend(news)
                if not quiet:
                    print(f'  ✓ {name}: +{len(news)}')
            except Exception as e:
                if not quiet:
                    print(f'  ⚠ {name}: 失败')
    
    save_cache({'news': results, 'time': time.time()})
    elapsed = time.time() - start
    if not quiet:
        print(f'  ⏱️ {elapsed:.1f}s | {len(results)}条')
    return results

def detect_hot_topics(news_list, top_n=5):
    stopwords = {'的','了','在','是','和','就','不','都','一','上','也','很','到','说','有','这','那','要','会','对','为','能','可','与','及','等'}
    word_freq = {}
    for n in news_list:
        words = re.findall(r'[\u4e00-\u9fff]{2,4}', n.get('title', ''))
        for w in words:
            if w not in stopwords and len(w) >= 2:
                word_freq[w] = word_freq.get(w, 0) + 1
    return sorted(word_freq.items(), key=lambda x: -x[1])[:top_n]
