def analyze(sym, live=False):
    sym = sym.upper()
    fin, analyst, price, beta, eps = fetch(sym)
    if live:
        p = fetch_live(sym)
        if p > 0: price = p
    
    eps_ttm = eps * 0.8
    eps_fwd = eps * (1 + fin[5])
    dcf = calc_dcf(eps_ttm, eps_fwd, beta, fin[3])
    score = calc_score(fin, price, eps, beta, analyst[0])
    upside = (analyst[0] - price) / price * 100
    if score >= 75 and upside >= 25: rating = "STRONG BUY"
    elif score >= 65 or upside >= 25: rating = "BUY"
    elif score >= 55 or upside >= 15: rating = "HOLD"
    elif score >= 45: rating = "UNDERPERFORM"
    else: rating = "SELL"
    
    pe = price / eps if eps > 0 else 50
    peg = pe / (fin[5] * 100) if fin[5] > 0 else 5
    pe_pct = min(100, max(0, (pe - 10) / 40 * 100))
    rec = "Deep Value" if pe_pct < 20 else "Attractive" if pe_pct < 40 else "Fair" if pe_pct < 60 else "Expensive" if pe_pct < 80 else "Premium"
    
    now = datetime.now()
    return {"symbol": sym, "price": price, "target": analyst[0], "upside": upside, "score": score,
            "rating": rating, "pe": pe, "fpe": fin[6] * 100, "peg": peg, "dcf_bull": dcf["Bull"],
            "dcf_base": dcf["Base"], "dcf_bear": dcf["Bear"], "roe": fin[2] * 100, "pm": fin[1] * 100,
            "de": fin[4], "fcf": fin[6] * 100, "div": fin[7] * 100, "beta": beta, "recommend": rec,
            "analyst_rating": analyst[1], "num_analysts": analyst[2], "rev_g": fin[5] * 100,
            "gm": fin[0] * 100, "roic": fin[3] * 100,
            "fetched_at": now.isoformat(), "price_source": "live" if live else "cached",
            "expires_at": (now.replace(second=0, microsecond=0).timestamp() + 300)}

def analyze_multiple(symbols, live=False):
    return [analyze(s, live) for s in symbols]
