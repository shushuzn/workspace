@lru_cache(maxsize=1024)
def get_importance(title: str) -> str:
    """判断新闻重要性 (带缓存)"""
    title_lower = title.lower()
    for level in ["critical", "high", "medium"]:
        keyword_set = IMPORTANCE_KEYWORD_SETS.get(level, set())
        if any(kw in title_lower for kw in keyword_set):
            return level
    return "low"


@lru_cache(maxsize=1024)
def get_sentiment_score(title: str) -> int:
    """情感评分 (带缓存)"""
    title_lower = title.lower()
    positive_count = sum(1 for w in POSITIVE_SET if w in title_lower)
    negative_count = sum(1 for w in NEGATIVE_SET if w in title_lower)
    score = (positive_count - negative_count) * 10
    return max(-100, min(100, score))


def analyze_sentiment(title: str) -> Tuple[str, int]:
    """情感分析，返回 (标签, 分数)"""
    score = get_sentiment_score(title)
    if score > 0:
        return "positive", score
    elif score < 0:
        return "negative", score
    return "neutral", 0
