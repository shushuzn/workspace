def cmd_fetch(args):
    now = datetime.now().strftime("%Y-%m-%d %H:%M")
    total_src = len(CONFIG["news"]["sources"])
    
    print(f"""
+--- NEWS FETCH @ {now} --------------------------------+
""")
    data = NewsData()
    total_new = 0
    sentiment_stats = {"利好": 0, "利空": 0, "中性": 0}
    source_stats = []
    
    # 使用进度条
    for idx, source in enumerate(CONFIG["news"]["sources"]):
        cat = source.get("category", "")
        pct = (idx + 1) / total_src * 100
        bar = "#" * int(pct / 5) + "-" * (20 - int(pct / 5))
        
        if source["type"] == "rss":
            items = fetch_rss(source["url"])
        else:
            items = fetch_json(source["url"])
        
        new_count = 0
        category = source.get("category", "")
        for item in items[:CONFIG["news"]["max_per_fetch"]]:
            if data.add_news(item["title"], item["url"], source["name"], category):
                new_count += 1
                total_new += 1
                sentiment, sector, keywords, codes = analyze(item["title"])
                sentiment_stats[sentiment] += 1
                icon = "[+]" if sentiment == "利好" else "[-]" if sentiment == "利空" else "[=]"
                kw_str = " #" + "#".join(keywords[:2]) if keywords else ""
                print(f"  {icon} {item['title'][:40]}{kw_str}")
        
        source_stats.append({"name": source["name"], "cat": cat, "new": new_count, "total": len(items)})
    
    # 统计摘要
    print(f"""
+--- FETCH SUMMARY ------------------------------------+
  Total New:  {total_new} items
  [+利好]:    {sentiment_stats['利好']:>3}  [-利空]: {sentiment_stats['利空']:>3}  [=中性]: {sentiment_stats['中性']:>3}
  ---------------------------------------------------""")
    
    for s in sorted(source_stats, key=lambda x: -x["new"])[:5]:
        if s["new"] > 0:
            print(f"  {s['cat']:<6} +{s['new']:<3} ({s['total']} fetched)")
    
    print("+----------------------------------------------------+")
    return total_new
