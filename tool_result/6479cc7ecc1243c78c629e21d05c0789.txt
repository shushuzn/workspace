#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
News to All - 推送到所有平台

用法:
    py news_to_all.py <title> <url>
    py news_to_all.py digest
    py news_to_all.py test
"""

import sys
import subprocess
from pathlib import Path

BASE_DIR = Path(r'D:\OpenClaw\workspace\30-scripts-tools')


def run_cmd(script, *args):
    """运行子命令"""
    cmd = ['py', str(BASE_DIR / script)] + list(args)
    result = subprocess.run(cmd, capture_output=True, text=True)
    return result.returncode == 0, result.stdout, result.stderr


def push_to_feishu(title, url):
    """推送到飞书"""
    try:
        # 使用 feishu_assistant 发送
        success, out, err = run_cmd('feishu_assistant.py', 'msg', f"📰 {title}\n\n{url}")
        if success:
            print("  [OK] Feishu")
            return True
        else:
            print(f"  [FAIL] Feishu: {err[:100]}")
            return False
    except Exception as e:
        print(f"  [FAIL] Feishu: {e}")
        return False


def push_to_qq(title, url):
    """推送到 QQ"""
    try:
        success, out, err = run_cmd('news_to_qq_napcat.py', 'send', title, url)
        if success and '[OK]' in out:
            print("  [OK] QQ")
            return True
        else:
            print(f"  [FAIL] QQ: {err[:100] if err else out[:100]}")
            return False
    except Exception as e:
        print(f"  [FAIL] QQ: {e}")
        return False


def cmd_send(args):
    """发送到所有平台"""
    if len(args) < 2:
        print("[ERROR] 用法: py news_to_all.py <title> <url>")
        return
    
    title = args[0]
    url = args[1]
    
    print(f"\n[PUSH] 推送新闻: {title[:40]}...")
    print("-" * 40)
    
    feishu_ok = push_to_feishu(title, url)
    qq_ok = push_to_qq(title, url)
    
    print("-" * 40)
    if feishu_ok and qq_ok:
        print("[OK] 全部推送成功")
    elif feishu_ok or qq_ok:
        print("[WARN] 部分推送成功")
    else:
        print("[FAIL] 全部推送失败")


def cmd_digest(args):
    """发送摘要到所有平台"""
    print("\n[PUSH] 推送新闻摘要")
    print("-" * 40)
    
    # QQ 摘要
    print("[QQ]")
    success, out, err = run_cmd('news_to_qq_napcat.py', 'digest', '5')
    if success:
        print("  [OK] QQ 摘要已发送")
    else:
        print(f"  [FAIL] {err[:100]}")
    
    # 飞书摘要
    print("[Feishu]")
    success, out, err = run_cmd('feishu_assistant.py', 'daily')
    if success:
        print("  [OK] 飞书摘要已发送")
    else:
        print(f"  [FAIL] {err[:100]}")


def cmd_test(args):
    """测试所有平台"""
    print("\n[TEST] 测试所有平台连接")
    print("-" * 40)
    
    # 测试 QQ
    print("[QQ]")
    success, out, err = run_cmd('news_to_qq_napcat.py', 'test')
    if success and '[OK]' in out:
        print("  [OK] QQ 连接正常")
    else:
        print(f"  [FAIL] {err[:100] if err else out[:100]}")
    
    # 测试飞书
    print("[Feishu]")
    success, out, err = run_cmd('feishu_assistant.py', 'msg', '🤖 OpenClaw 测试消息')
    if success:
        print("  [OK] 飞书连接正常")
    else:
        print(f"  [FAIL] {err[:100]}")


def main():
    if len(sys.argv) < 2:
        print(__doc__)
        return
    
    cmd = sys.argv[1].lower()
    args = sys.argv[2:]
    
    commands = {
        'send': cmd_send,
        'digest': cmd_digest,
        'test': cmd_test,
    }
    
    if cmd in commands:
        commands[cmd](args)
    else:
        print(f"[ERROR] 未知命令: {cmd}")
        print("使用: send, digest, test")


if __name__ == "__main__":
    main()
