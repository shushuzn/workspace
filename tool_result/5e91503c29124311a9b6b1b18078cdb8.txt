"""
Git Pre-Commit Hook - MANDATORY WORKFLOW CHECK
Blocks commit if workflow checks are not passed
"""
import subprocess
import sys
import os
from pathlib import Path

def run_check():
    print("\n" + "=" * 40)
    print("[LOCK] WORKFLOW GUARD - Pre-Commit Check")
    print("=" * 40)
    
    # Get staged .py files in 30-scripts-tools
    result = subprocess.run(
        ["git", "diff", "--cached", "--name-only"],
        capture_output=True, text=True
    )
    
    staged_files = [f for f in result.stdout.strip().split("\n") if f]
    tool_files = [f for f in staged_files if "30-scripts-tools" in f and f.endswith(".py")]
    
    if not tool_files:
        print("No tool files staged, allowing commit")
        return True
    
    for tool_file in tool_files:
        print(f"\nChecking: {tool_file}")
        
        # Step 1: Validate
        print("[1/2] Validating...")
        val_result = subprocess.run(
            [sys.executable, "30-scripts-tools/tool_validator_001.py", "--check", tool_file],
            capture_output=True, text=True
        )
        
        if "PASS" not in val_result.stdout or "errors\": 0" not in val_result.stdout:
            print("[X] VALIDATION FAILED")
            print(val_result.stdout[:200])
            return False
        
        # Step 2: Naming
        print("[2/2] Checking naming...")
        name_result = subprocess.run(
            [sys.executable, "30-scripts-tools/tool_namer_001.py", "--check", Path(tool_file).name],
            capture_output=True, text=True
        )
        
        if name_result.returncode != 0:
            print("[X] NAMING CHECK FAILED")
            print(name_result.stdout[:200])
            return False
    
    print("\n[V] ALL CHECKS PASSED")
    print("[V] Commit allowed")
    return True

if __name__ == "__main__":
    if not run_check():
        sys.exit(1)
