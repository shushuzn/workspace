def get_market_sentiment(news_list):
    """计算市场整体情绪"""
    if not news_list:
        return None
    
    positive = sum(1 for n in news_list if n.get('llm_sentiment') == 'positive')
    negative = sum(1 for n in news_list if n.get('llm_sentiment') == 'negative')
    neutral = len(news_list) - positive - negative
    
    total = len(news_list)
    avg_score = sum(n.get('llm_score', 0) for n in news_list) / total if total > 0 else 0
    
    # 判断整体情绪
    if avg_score > 15:
        overall = 'bullish'
    elif avg_score < -15:
        overall = 'bearish'
    else:
        overall = 'neutral'
    
    # 市场影响统计
    market_counts = {}
    for n in news_list:
        for market in n.get('llm_markets', []):
            market_counts[market] = market_counts.get(market, 0) + 1
    
    # 投资建议统计
    advice_counts = {}
    for n in news_list:
        advice = n.get('llm_advice', 'neutral')
        advice_counts[advice] = advice_counts.get(advice, 0) + 1
    
    return {
        'positive': positive,
        'negative': negative,
        'neutral': neutral,
        'avg_score': avg_score,
