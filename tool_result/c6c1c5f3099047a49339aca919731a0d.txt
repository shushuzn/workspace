C = {'bg': '#0f0f23', 'card': 'rgba(30,30,60,0.8)', 'border': 'rgba(255,255,255,0.1)',

def g(): return {'padding': '14px 24px', 'color': C['muted'], 'cursor': 'pointer', 
                'fontSize': '14px', 'fontWeight': '500', 'borderLeft': '3px solid transparent'}
def ga(): return {**g(), 'color': C['primary'], 'background': 'rgba(99,102,241,0.1)', 
                 'borderLeft': f'3px solid {C["primary"]}'}

def fmt(n, d=2): 
    if n is None: return "N/A"
    try: return f"{n:,.{d}f}"
    except: return str(n)

def sc(s):
    return '#059669' if s>=80 else '#10b981' if s>=70 else '#f59e0b' if s>=50 else '#f97316' if s>=30 else '#ef4444'

def sl(s):
    return "STRONG BUY" if s>=80 else "BUY" if s>=70 else "HOLD" if s>=50 else "SELL" if s>=30 else "STRONG SELL"

def card(content, style=None):
    base = {'background': C['card'], 'backdropFilter': 'blur(16px)', 'borderRadius': '16px',
            'border': f'1px solid {C["border"]}', 'padding': '24px', 'marginBottom': '20px'}
    if style: base.update(style)
    return html.Div(content, style=base)

def stat(title, val, chg=None, icon=None, acc=None):
    chg_str, chg_clr = "", C['muted']
    if chg is not None:
        chg_str = f"{'↑' if chg>=0 else '↓'} {abs(chg):.2f}%"
        chg_clr = C['bull'] if chg>=0 else C['bear']
    top = f'3px solid {acc}' if acc else None
    return card([
        html.Div([html.Div([html.Div(title, style={'fontSize':'11px','color':C['muted'],'textTransform':'uppercase','letterSpacing':'0.5px'}),
                           html.Div(val, style={'fontSize':'26px','fontWeight':'700','marginTop':'8px'}),
                           html.Div(chg_str, style={'fontSize':'13px','marginTop':'4px','color':chg_clr}) if chg is not None else html.Div()], style={'width':'100%'}),
                  html.Div(icon, style={'fontSize':'30px','opacity':'0.7'}) if icon else html.Div()], 
                 style={'display':'flex','justifyContent':'space-between','alignItems':'flex-start'})
    ], {'borderTop': top} if top else {})

def stock_card(sym, d):
    if not d or 'error' in d:
        return card([html.Div(f"❌ {sym}", style={'textAlign':'center','padding':'20px'}),
                    html.Div(d.get('error','No data') if d else 'N/A', 
                            style={'textAlign':'center','color':C['muted'],'fontSize':'12px'})])
    score = d.get('score', 50)
    price = d.get('price', 0)
    upside = d.get('upside', 0)
    clr = sc(score)
    return card([
        html.Div([html.Div([html.Div(sym, style={'fontSize':'18px','fontWeight':'700'}),
                           html.Div(d.get('name','')[:20], style={'fontSize':'11px','color':C['muted'],'marginTop':'2px'})]),
                  html.Div([html.Div(f"${fmt(price)}", style={'fontSize':'20px','fontWeight':'700'}),
                           html.Div(f"{upside:+.1f}%", style={'fontSize':'12px','color':C['bull'] if upside>0 else C['bear']})],
                           style={'textAlign':'right'})],
                 style={'display':'flex','justifyContent':'space-between','marginBottom':'20px'}),
        html.Div([html.Div(sl(score), style={'fontSize':'10px','fontWeight':'700','color':clr,'textAlign':'center','marginBottom':'6px'}),
                  html.Div([html.Div('', style={'height':'6px','borderRadius':'3px',
                              'background':f'linear-gradient(90deg, {clr} {score}%, rgba(255,255,255,0.1) {score}%)'})],
                           style={'background':'rgba(255,255,255,0.1)','borderRadius':'3px'})],
                 style={'marginBottom':'16px'}),
        html.Div([html.Div([html.Div(k.upper(), style={'fontSize':'9px','color':C['muted']}),
                           html.Div(fmt(v), style={'fontSize':'13px','fontWeight':'600'})],
                          style={'textAlign':'center'}) for k,v in [('P/E',d.get('pe',0)),('EPS',d.get('eps',0)),
                          ('Mkt',d.get('marketCap',0)),('Div',d.get('dividendYield',0)*100 if d.get('dividendYield') else 0)]],
                 style={'display':'grid','gridTemplateColumns':'repeat(4,1fr)','gap':'12px','marginBottom':'16px',
                        'padding':'12px','background':'rgba(0,0,0,0.2)','borderRadius':'10px'}),
        html.Div([html.Button('📊 Details', n_clicks=0, style={'flex':'1','marginRight':'8px','background':'transparent',
                   'border':f'1px solid {C["border"]}','color':C['text'],'padding':'10px','borderRadius':'8px','cursor':'pointer'}),
                  html.Button('🔄 Refresh', n_clicks=0, style={'flex':'1','background':'transparent',
                   'border':f'1px solid {C["border"]}','color':C['text'],'padding':'10px','borderRadius':'8px','cursor':'pointer'})],
                 style={'display':'flex'})
    ])

def price_chart(sym):
    from stock_pro.core import fetch
    meta, data, err = fetch(sym)
    if err: return html.Div(f"Error: {err}", style={'textAlign':'center','padding':'60px','color':C['muted']})
    
    ts = data.get('timestamp', [])
    closes = data.get('indicators',{}).get('quote',[{}])[0].get('close',[])
    highs = data.get('indicators',{}).get('quote',[{}])[0].get('high',[])
    lows = data.get('indicators',{}).get('quote',[{}])[0].get('low',[])
    opens = data.get('indicators',{}).get('quote',[{}])[0].get('open',[])
    
    if not closes: return html.Div("No data", style={'textAlign':'center','padding':'60px'})
    
    # Line + Volume chart
    from plotly.subplots import make_subplots
    fig = make_subplots(rows=2, cols=1, shared_xaxes=True, vertical_spacing=0.1,
                       row_heights=[0.7, 0.3])
    
    fig.add_trace(go.Scatter(x=list(range(len(closes))), y=closes, mode='lines+markers',
                             line=dict(color=C['primary'], width=2), name='Price', fill='tozeroy',
                             fillcolor='rgba(99,102,241,0.1)'), row=1, col=1
