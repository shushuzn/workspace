def fetch_all(quiet=False, dedup=True):
    cache = load_cache()
    now = time.time()
    if cache.get('news') and now - cache.get('time', 0) < CACHE_TTL:
        if not quiet:
            elapsed = int((now - cache.get('time', 0)) / 60)
            print(f'  💾 缓存: {len(cache["news"])} 条 ({elapsed}m前)')
        return cache['news']

    results = []
    start = now
    if not quiet:
        print('  ▶ 并行抓取...')

    with ThreadPoolExecutor(max_workers=FETCH_WORKERS) as executor:
        futures = {executor.submit(fetcher): name for name, fetcher in SOURCES}
        for future in as_completed(futures):
            name = futures[future]
            try:
                news = future.result()
                results.extend(news)
                if not quiet:
                    print(f'  ✓ {name}: +{len(news)}')
            except Exception:
                if not quiet:
                    print(f'  ⚠ {name}: 失败')

    if dedup:
        original_count = len(results)
        results = dedup_news(results)
        if not quiet and original_count != len(results):
            print(f'  🔃 去重: {original_count} → {len(results)}')

    save_cache({'news': results, 'time': time.time()})
    elapsed = time.time() - start
    if not quiet:
        print(f'  ⏱️ {elapsed:.1f}s | {len(results)}条')
    return results
