def fetch_all():
    config = load_config()
    all_news = []
    sources = {}
    
    banner("NewsHub Fetcher")
    info(f"Time: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print()
    
    for key, src in config.get('news_sources', {}).items():
        if not src.get('enable', True):
            continue
        
        name = src.get('name', key)
        info(f"Fetching {name}...")
        
        if key == 'jin10':
            news = fetch_jin10(src.get('limit', 20))
        else:
            news = fetch_rss(src['url'], name)
        
        sources[name] = len(news)
        all_news.extend(news)
        
        if news:
            ok(f"{name}: +{len(news)}")
        else:
            warn(f"{name}: 0")
    
    all_news.sort(key=lambda x: x.get('ts', 0), reverse=True)
    summary(len(all_news), sources)
    
    return all_news
