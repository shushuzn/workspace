class S:
    HEADER, BLUE, CYAN, GREEN, YELLOW, RED, BOLD, END = "\033[95m", "\033[94m", "\033[96m", "\033[92m", "\033[93m", "\033[91m", "\033[1m", "\033[0m"

def c(s, t): return f"{s}{t}{S.END}" if sys.stdout.isatty() else t
def B(t): return c(S.BOLD, t)
def R(t): return c(S.RED, t)
def G(t): return c(S.GREEN, t)
def Y(t): return c(S.YELLOW, t)
def L(t): return c(S.BLUE, t)
def C(t): return c(S.CYAN, t)

def info(m): print(f"  {L('▶')} {m}")
def ok(m): print(f"  {G('✓')} {G(m)}")
def warn(m): print(f"  {Y('⚠')} {Y(m)}")
def error(m): print(f"  {R('✗')} {R(m)}")

def stats_box(cnt, new, srcs):
    print()
    print(C("  ┌─") + f" 新闻统计 {C('─'*32)}┐")
    print(C("  │") + f"  📰 总数: {cnt:<6}  🆕 新: {new:<6}  {C('│')}")
    print(C("  ├") + f" {C('─'*44)} {C('├')}")
    for s, n in srcs.items():
        icon = "🔹" if "金十" in s else ("🔸" if "Investing" in s else "🔶")
        bar = "█" * min(n // 2, 10)
        print(C("  │") + f"  {icon} {s:<12} {n:>2} 条 {bar:<10} {C('│')}")
    print(C("  └") + f" {C('─'*44)} {C('┘')}")
    print()

def result_box(mode, sent, total, elapsed, cached=False):
    print()
    print(C("  ┌─") + f" 结果 {C('─'*40)}┐")
    print(C("  │") + f"  📊 Mode: {mode:<10}  📤 Sent: {sent}/{total}      {C('│')}")
    cache_tag = " ✅" if cached else ""
    print(C("  │") + f"  ⏱️ 耗时: {elapsed:.1f}s{cache_tag:<20} {C('│')}")
    print(C("  └") + f" {C('─'*46)} {C('┘')}")
    print()

def bar(cur, tot, w=25):
    f = int(w * cur / tot) if tot > 0 else 0
    return f"[{'█'*f}{'─'*(w-f)}] {cur}/{tot}"

def prog(cur, tot, pre=""):
    print(f"\r  {pre}{bar(cur, tot)}", end="", flush=True)

def compact_view(news_list, limit=10):
    """紧凑视图"""
    for i, n in enumerate(news_list[:limit], 1):
        title = n['title'][:45] + "..." if len(n['title']) > 45 else n['title']
        s = n.get('source', '')
        icon = "📰" if "金十" in s else ("💹" if "股票" in s else ("🌐" if "Investing" in s else ("📊" if "Seeking" in s else "📱")))
        print(f"  {i:>2}. {icon} {title}")
    if len(news_list) > limit:
        print(f"  ... 还有 {len(news_list) - limit} 条")
