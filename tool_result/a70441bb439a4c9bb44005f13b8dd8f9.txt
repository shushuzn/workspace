def build_digest_card(news_list, template=None):
    tmpl = template or DEFAULT_TEMPLATES["digest"]
    items = []
    for i, n in enumerate(news_list[:tmpl.get("max_items", 6)], 1):
        title = n["title"][:50] + "…" if len(n["title"]) > 50 else n["title"]
        source = n.get("source", "")
        if tmpl.get("show_sentiment", True):
            score = n.get("score", 0)
            sent_icon = "🟢" if score > 20 else ("🔴" if score < -20 else "⚪")
        else:
            sent_icon = ""
        items.append({"tag": "div", "text": {"tag": "lark_md", "content": f"{sent_icon} {i}. {title}\n   📰 {source}"}})

    header = {"tag": "div", "text": {"tag": "plain_text", "content": "📊 今日要闻" if tmpl.get("show_sentiment", True) else "📰 新闻摘要"}}
    elements = [header] + items + [{"tag": "hr"}]
    return {"config": {"wide_screen_mode": True}, "elements": elements}
