def filter_duplicates(items, history):
    """过滤重复"""
    new_items = []
    for item in items:
        if not is_duplicate(item["title"], history):
            item["hash"] = hash_text(item["title"])
            new_items.append(item)
    return new_items


def check_keywords(items, keywords):
    """检查关键词匹配"""
    alerts = []
    for item in items:
        for kw in keywords:
            if kw.lower() in item["title"].lower():
                item["keyword"] = kw
                alerts.append(item)
                break
    return alerts
