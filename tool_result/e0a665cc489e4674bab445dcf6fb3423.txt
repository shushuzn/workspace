#!/usr/bin/env python3
"""
NewsHub RSS Server - 本地RSS服务
提供缓存新闻的RSS订阅
"""
import sys
sys.stdout.reconfigure(encoding='utf-8')
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
from news_fetcher import fetch_all, load_cache
from datetime import datetime
import json

from http.server import HTTPServer, BaseHTTPRequestHandler
import xml.etree.ElementTree as ET

PORT = 8888

def news_to_rss(news_list, base_url="http://localhost:8888"):
    """将新闻列表转换为RSS XML"""
    root = ET.Element('rss', version='2.0')
    channel = ET.SubElement(root, 'channel')

    # 频道信息
    ET.SubElement(channel, 'title').text = 'NewsHub - 金融新闻聚合'
    ET.SubElement(channel, 'link').text = 'https://github.com/OpenClaw/NewsHub'
    ET.SubElement(channel, 'description').text = f'聚合 {len(news_list)} 条新闻'
    ET.SubElement(channel, 'language').text = 'zh-cn'
    ET.SubElement(channel, 'lastBuildDate').text = datetime.now().strftime('%a, %d %b %Y %H:%M:%S GMT')

    for n in news_list[:50]:
        item = ET.SubElement(channel, 'item')
        ET.SubElement(item, 'title').text = n.get('title', '')
        ET.SubElement(item, 'link').text = n.get('url', '')
        ET.SubElement(item, 'description').text = f"来源: {n.get('source', '')}"
        ET.SubElement(item, 'pubDate').text = datetime.fromtimestamp(n.get('ts', 0)).strftime('%a, %d %b %Y %H:%M:%S GMT')

    return '<?xml version="1.0" encoding="UTF-8"?>\n' + ET.tostring(root, encoding='unicode')

def news_to_html(news_list):
    """将新闻列表转换为HTML"""
    # 分类统计
    cat_colors = {"stock": "#00d4ff", "forex": "#ffd700", "commodity": "#ff6b35", "macro": "#9b59b6", "trade": "#e74c3c", "tech": "#2ecc71", "china": "#e74c3c", "us": "#3498db"}
    cat_names = {"stock": "股市", "forex": "外汇", "commodity": "大宗商品", "macro": "宏观", "trade": "贸易", "tech": "科技", "china": "中国", "us": "美国"}

    cat_stats = {}
    sent_stats = {"positive": 0, "negative": 0, "neutral": 0}
    for n in news_list:
        cat = n.get('category', 'other')
        cat_stats[cat] = cat_stats.get(cat, 0) + 1
        sent = n.get('sentiment', 'neutral')
        sent_stats[sent] = sent_stats.get(sent, 0) + 1

    stats_html = '<div style="display:flex;gap:10px;margin-bottom:20px;flex-wrap:wrap;">'
    for cat, cnt in sorted(cat_stats.items(), key=lambda x: -x[1]):
        color = cat_colors.get(cat, "#888")
        name = cat_names.get(cat, cat)
        stats_html += f'<span style="background:{color};padding:5px 10px;border-radius:5px;font-size:0.9em;">{name}: {cnt}</span>'
    stats_html += '</div>'

    sentiment_html = f'<div style="margin-bottom:20px;">'
    sentiment_html += f'<span style="color:#2ecc71;">🟢 利好: {sent_stats["positive"]}</span> '
    sentiment_html += f'<span style="color:#e74c3c;">🔴 利空: {sent_stats["negative"]}</span> '
    sentiment_html += f'<span style="color:#888;">⚪ 中性: {sent_stats["neutral"]}</span>'
    sentiment_html += '</div>'

    html = f"""<!DOCTYPE html>
<html lang="zh">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>NewsHub - {len(news_list)} 条新闻</title>
    <style>
        body {{ font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif; max-width: 900px; margin: 0 auto; padding: 20px; background: #1a1a2e; color: #eee; }}
        h1 {{ color: #00d4ff; border-bottom: 2px solid #00d4ff; padding-bottom: 10px; }}
        .meta {{ color: #888; font-size: 0.9em; margin-bottom: 20px; }}
        .news {{ background: #16213e; border-radius: 8px; padding: 15px; margin: 10px 0; border-left: 4px solid #00d4ff; }}
        .news h3 {{ margin: 0 0 8px 0; color: #fff; }}
        .news a {{ color: #00d4ff; text-decoration: none; }}
        .news a:hover {{ text-decoration: underline; }}
        .news .source {{ color: #888; font-size: 0.85em; margin-top: 5px; }}
        .tag {{ display: inline-block; padding: 2px 8px; border-radius: 4px; font-size: 0.75em; margin-right: 5px; }}
        .rss-link {{ background: #ff6b35; color: white; padding: 10px 20px; border-radius: 5px; text-decoration: none; display: inline-block; margin-bottom: 20px; }}
        .positive {{ border-color: #2ecc71; }}
        .negative {{ border-color: #e74c3c; }}
    </style>
</head>
<body>
    <h1>📊 NewsHub</h1>
    <div class="meta">共 {len(news_list)} 条新闻 | 更新时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}</div>
    <a class="rss-link" href="/news.xml">📡 RSS订阅</a>
    {sentiment_html}
    {stats_html}
"""

    for n in news_list[:30]:
        source = n.get('source', '')
        source_icon = "📰" if "新浪" in source else ("🌐" if "Investing" in source else ("📊" if "Seeking" in source else "📱"))
        cat = n.get('category', 'other')
        sent = n.get('sentiment', '')
        cat_color = cat_colors.get(cat, '#888')
        cat_name = cat_names.get(cat, cat)

        sent_icon = "🟢" if sent == 'positive' else ("🔴" if sent == 'negative' else "")

        news_class = "positive" if sent == 'positive' else ("negative" if sent == 'negative' else "")

        html += f"""
    <div class="news {news_class}">
        <h3>{source_icon} {sent_icon} {n.get('title', '')}</h3>
        <span class="tag" style="background:{cat_color};">{cat_name}</span>
        <a href="{n.get('url', '')}" target="_blank">🔗 查看详情</a>
        <div class="source">来源: {source}</div>
    </div>
"""

    html += """
</body>
</html>"""
    return html

class NewsHandler(BaseHTTPRequestHandler):
    def log_message(self, format, *args):
        pass  # 静默日志

    def do_GET(self):
        if self.path == '/' or self.path == '/index.html':
            news = fetch_all(quiet=True, use_cache=True)
            html = news_to_html(news)
            self.send_response(200)
            self.send_header('Content-Type', 'text/html; charset=utf-8')
            self.end_headers()
            self.wfile.write(html.encode('utf-8'))

        elif self.path == '/news.xml' or self.path == '/rss':
            news = fetch_all(quiet=True, use_cache=True)
            rss = news_to_rss(news)
            self.send_response(200)
            self.send_header('Content-Type', 'application/xml; charset=utf-8')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(rss.encode('utf-8'))

        elif self.path == '/api/news':
            news = fetch_all(quiet=True, use_cache=True)
            self.send_response(200)
            self.send_header('Content-Type', 'application/json; charset=utf-8')
            self.end_headers()
            self.wfile.write(json.dumps(news, ensure_ascii=False).encode('utf-8'))

        elif self.path == '/api/stats':
            news = fetch_all(quiet=True, use_cache=True)
            sources = {}
            for n in news:
                s = n.get('source', 'Other')
                sources[s] = sources.get(s, 0) + 1
            stats = {"total": len(news), "sources": sources, "time": datetime.now().isoformat()}
            self.send_response(200)
            self.send_header('Content-Type', 'application/json; charset=utf-8')
            self.end_headers()
            self.wfile.write(json.dumps(stats, ensure_ascii=False).encode('utf-8'))

        else:
            self.send_response(404)
            self.end_headers()
            self.wfile.write(b'Not Found')

    def do_OPTIONS(self):
        self.send_response(200)
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()

def run_server(port=PORT):
    server = HTTPServer(('0.0.0.0', port), NewsHandler)
    print(f"""
╔════════════════════════════════════════════════╗
║  📡 NewsHub RSS Server                        ║
╚════════════════════════════════════════════════╝
  
  🌐 http://localhost:{port}         (HTML)
  📡 http://localhost:{port}/news.xml   (RSS)
  📊 http://localhost:{port}/api/stats  (JSON)
  
  按 Ctrl+C 停止服务
""")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\n\n  👋 Server stopped")
        server.shutdown()

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument('--port', type=int, default=PORT, help='端口')
    args = parser.parse_args()

    run_server(args.port)
