def fetch_sina_news(limit=20):
    """新浪财经"""
    news = []
    try:
        url = f'https://feed.mix.sina.com.cn/api/roll/get?pageid=153&lid=2514&num={limit}&versionNumber=1.2.4&page=1'
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=10, context=get_ssl()) as r:
            data = json.loads(r.read().decode('utf-8'))
        
        items = data.get('result', {}).get('data', [])
        for item in items[:limit]:
            title = item.get('title', '').strip()
            if title:
                news.append({
                    "title": title,
                    "url": item.get('url', ''),
                    "source": "新浪财经",
                    "category": "finance",
                    "ts": time.time()
                })
    except Exception as e:
        error(f"Sina: {e}")
    return news
