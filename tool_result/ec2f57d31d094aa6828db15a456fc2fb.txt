    def deduplicate(self, news_list: List[NewsItem]) -> List[NewsItem]:
        unique = []
        new_hashes = set()

        for news in news_list:
            if news.hash not in self.existing_hashes and news.hash not in new_hashes:
                unique.append(news)
                new_hashes.add(news.hash)

        return unique
