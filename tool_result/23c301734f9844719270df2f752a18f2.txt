# ========== 样式 ==========
class S:
    HEADER = "\033[95m"
    BLUE = "\033[94m"
    CYAN = "\033[96m"
    GREEN = "\033[92m"
    YELLOW = "\033[93m"
    RED = "\033[91m"
    BOLD = "\033[1m"
    END = "\033[0m"


def c(style: str, text: str) -> str:
    """应用样式"""
    return f"{style}{text}{S.END}" if sys.stdout.isatty() else text


def B(text: str) -> str:
    return c(S.BOLD, text)


def G(text: str) -> str:
    return c(S.GREEN, text)


def Y(text: str) -> str:
    return c(S.YELLOW, text)


def L(text: str) -> str:
    return c(S.BLUE, text)


def C(text: str) -> str:
    return c(S.CYAN, text)


def R(text: str) -> str:
    return c(S.RED, text)


def info(msg: str):
    print(f"  {L('▶')} {msg}")


def ok(msg: str):
    print(f"  {G('✓')} {G(msg)}")


def warn(msg: str):
    print(f"  {Y('⚠')} {Y(msg)}")


def error(msg: str):
    print(f"  {R('✗')} {R(msg)}")


def banner(title: str, icon: str = "📡", width: int = 48):
    print()
    border = "═" * width
    print(C(f"╔{border}╗"))
    print(C("║") + f"  {icon} {B(title)}".ljust(width) + C("║"))
    print(C(f"╚{border}╝"))
    print()

def load_json_file(filepath: Path, default: Any = None) -> Any:
    """通用 JSON 加载"""
    if filepath.exists():
        try:
            with open(filepath, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            pass
    return default if default is not None else {}
