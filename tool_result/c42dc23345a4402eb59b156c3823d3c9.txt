#!/usr/bin/env python3
"""
NewsHub Scheduler - 定时任务
自动抓取新闻并推送到飞书
"""
import sys
import time
import subprocess
import json
from pathlib import Path
from datetime import datetime

sys.stdout.reconfigure(encoding='utf-8')
BASE_DIR = Path(__file__).parent
LOG_FILE = BASE_DIR / ".scheduler.log"
STATE_FILE = BASE_DIR / ".scheduler_state.json"

class Scheduler:
    def __init__(self):
        self.load_state()
    
    def load_state(self):
        if STATE_FILE.exists():
            with open(STATE_FILE, 'r', encoding='utf-8') as f:
                self.state = json.load(f)
        else:
            self.state = {
                "last_fetch": None,
                "last_digest": None,
                "last_alert": None,
                "fetch_count": 0
            }
    
    def save_state(self):
        with open(STATE_FILE, 'w', encoding='utf-8') as f:
            json.dump(self.state, f, ensure_ascii=False, indent=2)
    
    def log(self, msg):
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        log_line = f"[{timestamp}] {msg}"
        print(log_line)
        with open(LOG_FILE, 'a', encoding='utf-8') as f:
            f.write(log_line + "\n")
    
    def run_command(self, script, *args):
        cmd = [sys.executable, BASE_DIR / script, *args]
        result = subprocess.run(cmd, capture_output=True, text=True)
        return result.returncode == 0
    
    def should_run(self, task, interval_minutes):
        last = self.state.get(f"last_{task}")
        if not last:
            return True
        last_time = datetime.fromisoformat(last)
        elapsed = (datetime.now() - last_time).total_seconds() / 60
        return elapsed >= interval_minutes
    
    def mark_done(self, task):
        self.state[f"last_{task}"] = datetime.now().isoformat()
        self.state["fetch_count"] = self.state.get("fetch_count", 0) + 1
        self.save_state()
