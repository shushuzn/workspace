def cmd_status(args):
    data = NewsData()
    total = len(data.data["news"])
    unsent = len([n for n in data.data["news"] if not n.get("sent")])
    sent = total - unsent
    
    # 统计
    sectors = {}
    sentiments = {"利好": 0, "利空": 0, "中性": 0}
    for n in data.data["news"]:
        sectors[n.get("sector","通用")] = sectors.get(n.get("sector","通用"), 0) + 1
        sentiments[n.get("sentiment","中性")] = sentiments.get(n.get("sentiment","中性"), 0) + 1
    
    # 格式化
    now = datetime.now().strftime("%m-%d %H:%M")
    sent_pct = sent / total * 100 if total > 0 else 0
    
    # 头部
    print(f"\n{'='*34}")
    print(f"{c(Colors.BOLD, ' STATUS ' + now)}")
    print(f"{'='*34}")
    
    # 核心指标 - 单行显示
    print(f"\n{c(Colors.BOLD, 'News')} {total} | {c(Colors.BOLD, 'Sent')} {sent}/{total} ({sent_pct:.0f}%)")
    
    # 进度条
    bar_len = 20
    filled = int(sent_pct / 100 * bar_len)
    bar = c(Colors.GREEN, chr(9608) * filled) + c(Colors.DIM, chr(9617) * (bar_len - filled))
    print(f"[{bar}]")
    
    # 情绪分布
    pos = sentiments['利好']
    neg = sentiments['利空']
    neu = sentiments['中性']
    bar_len2 = 16
    pos_bar = int(pos / total * bar_len2) if total > 0 else 0
    neg_bar = int(neg / total * bar_len2) if total > 0 else 0
    sentiment_bar = cp(chr(9608) * pos_bar) + cn(chr(9608) * neg_bar) + cne(chr(9608) * (bar_len2 - pos_bar - neg_bar))
    print(f"\n{c(Colors.BOLD, 'Sentiment')}")
    print(f"[{sentiment_bar}]")
    print(f" {cp('+' + str(pos))} {cn('-' + str(neg))} {cne('=' + str(neu))}")
    
    # 风险评估
    risk_score = neg * 2 - pos
    if risk_score > 10:
        risk_text = cn("HIGH")
