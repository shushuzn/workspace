def cmd_multigroup(receive_ids=None):
    banner("多群推送")
    config = load_feishu_config()
    if not config.get("app_id"):
        error("未配置飞书应用")
        return

    receive_ids = receive_ids or config.get("group_ids", [])
    if not receive_ids:
        error("未配置群ID")
        return

    from async_fetcher import fetch_all
    news = fetch_all(quiet=True)[:6]
    card = build_digest_card(news)
    token = get_access_token()

    success = 0
    for rid in receive_ids:
        if send_card(token, rid, card).get("code") == 0:
            success += 1
        time.sleep(0.3)

    ok(f"已发送到 {success}/{len(receive_ids)} 个群")
    add_log("multigroup", success, len(news))
