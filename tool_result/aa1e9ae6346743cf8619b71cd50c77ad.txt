def build_news_item(title, url, source, timestamp=None):
    """统一构建新闻对象，减少各抓取源重复字段拼装。

    这里把时间戳提升到调用方批次级生成，避免同一批数据里频繁调用 time.time()。
    如果后面要加更多“手感参数”（比如优先级、抓取耗时），也只需要改这一处。
    """
    item = {'title': title, 'url': url, 'source': source, 'ts': timestamp or time.time()}
    item['category'] = classify_news(title)
    item['sentiment'], item['score'] = analyze_sentiment(title)
    return item

def classify_news(title):
    title_lower = title.lower()
    for cat, keywords in CATEGORIES.items():
        if any(kw.lower() in title_lower for kw in keywords):
            return cat
    return 'other'

def analyze_sentiment(title):
    title_lower = title.lower()
    pos = sum(1 for w in POSITIVE if w in title_lower)
    neg = sum(1 for w in NEGATIVE if w in title_lower)
    if pos > neg:
        return 'positive', 20
    elif neg > pos:
        return 'negative', -20
    return 'neutral', 0

def fetch_sina():
    news = []
    try:
        fetched_at = time.time()

            timeout=REQUEST_TIMEOUT,
            context=get_ssl_context(),
        )
        for item in data.get('result', {}).get('data', [])[:15]:
            title = item.get('title', '').strip()
            if title:
                news.append(build_news_item(title, item.get('url', ''), '新浪财经', fetched_at))
    except Exception as e:
        print(f'  ⚠ 新浪失败: {e}')
    return news
