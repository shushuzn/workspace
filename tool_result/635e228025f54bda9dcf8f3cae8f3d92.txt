# ========== 主函数 ==========
async def async_fetch_news(quiet: bool = False) -> List[dict]:
    config = load_config()
    settings = config.get("settings", {})

    fetcher = AsyncNewsFetcher(
        timeout=settings.get("fetch_timeout", 15),
        max_concurrent=settings.get("max_concurrent", 8),
        retry_count=settings.get("retry_count", 3),
        retry_delay=settings.get("retry_delay", 0.5),
        rate_limit=settings.get("rate_limit", 30),
        verify_ssl=settings.get("verify_ssl", True),  # SSL 验证 (默认开启)
    )

    start_time = time.time()
    news_list = await fetcher.fetch_all_sources()
    elapsed = time.time() - start_time

    unique_news = fetcher.deduplicate(news_list)
    enriched = fetcher.enrich(unique_news)

    all_hashes = fetcher.existing_hashes | {n.hash for n in enriched}
    save_hashes(all_hashes)

    if not quiet:
        print(f"\n  ⏱️ 耗时: {elapsed:.1f}s")
        print(f"  📰 总数: {len(news_list)} → 去重后: {len(enriched)}")

    try:
        cache_data = {
            "updated": datetime.now().isoformat(),
            "total": len(enriched),
            "news": [n.to_dict() for n in enriched],
        }
        with open(CACHE_FILE, "w", encoding="utf-8") as f:
            json.dump(cache_data, f, ensure_ascii=False, indent=2)
    except Exception as e:
        logger.error(f"缓存保存失败: {e}")

    return [n.to_dict() for n in enriched]


def fetch_all(quiet: bool = False) -> List[dict]:
    return asyncio.run(async_fetch_news(quiet))


if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="异步新闻抓取")
    parser.add_argument("--quiet", "-q", action="store_true", help="静默模式")
    parser.add_argument("--benchmark", "-b", action="store_true", help="性能测试")

    args = parser.parse_args()

    if args.benchmark:
        print("\n  🔬 性能测试")
        print("  " + "─" * 40)

        start = time.time()
        news = fetch_all(quiet=True)
        async_time = time.time() - start

        print(f"\n  异步版本: {async_time:.2f}s ({len(news)} 条)")
        print(f"  平均每源: {async_time / 9:.2f}s")
    else:
        fetch_all(quiet=args.quiet)
