def cmd_daily():
    banner("日报", "📊")
    all_news = fetch_all()
    history = load_history()
    info(f"获取到 {len(all_news)} 条新闻")
    if not all_news:
        warn("无新闻")
        return 0
    token = get_access_token()
    if not token:
        error("认证失败")
        return 0
    feishu = load_feishu_config()
    card = build_daily_summary(all_news)
    result = send_card(token, feishu.get('default_receive_id', ''), card)
    if result.get('code') == 0:
        for n in all_news:
            history.append(n['title'])
        save_history(history)
        ok(f"日报已发送 ({len(all_news)} 条)")
        add_log('daily', 1, len(all_news))
        return 1
    else:
        error(f"发送失败: {result.get('msg', 'Unknown')}")
        return 0
