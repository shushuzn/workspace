def rating(upside):
    if upside > 30: return "STRONG_BUY"
    if upside > 15: return "BUY"
    if upside > 0: return "HOLD"
    return "SELL"

def risk_level(q, fin):
    risk = q["beta"] * 20 + fin["de"] * 10
    if risk < 30: return "LOW"
    if risk < 50: return "MEDIUM"
    return "HIGH"
