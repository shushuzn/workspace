def cmd_summary():
    """发送摘要卡片"""
    banner("摘要卡片", "📊")
    all_news = fetch_all(quiet=True)
    
    info(f"新闻: {len(all_news)} 条")
    
    token = get_access_token()
    if not token:
        error("认证失败")
        return 0
    
    feishu = load_feishu_config()
    card = build_summary_card(all_news)
    result = send_card(token, feishu.get('default_receive_id', ''), card)
    
    if result.get('code') == 0:
        ok("摘要已发送")
        add_log('summary', 1, len(all_news))
        return 1
    else:
        error(f"发送失败: {result.get('msg', 'Unknown')}")
        return 0
