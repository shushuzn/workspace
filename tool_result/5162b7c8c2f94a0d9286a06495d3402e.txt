class NewsData:
    def __init__(self):
        self.data = self._load()
    
    def _load(self) -> Dict:
        if DATA_FILE.exists():
            try:
                with open(DATA_FILE, 'r', encoding='utf-8') as f:
                    return json.load(f)
            except:
                pass
        return {"news": [], "sent": [], "stats": {"total": 0, "qq": 0, "feishu": 0, "last_fetch": None}}
    
    def save(self):
        with open(DATA_FILE, 'w', encoding='utf-8') as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)
    
    def add_news(self, title: str, url: str, source: str = ""):
        news_id = hash(title + url) & 0xFFFFFFFF
        for n in self.data["news"]:
            if n["id"] == news_id:
                return False
        
        self.data["news"].append({
            "id": news_id,
            "title": title,
            "url": url,
            "source": source,
            "fetched_at": datetime.now().isoformat(),
            "sent": False
        })
        self.data["stats"]["total"] += 1
