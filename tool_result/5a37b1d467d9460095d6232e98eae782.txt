@app.route('/')
def index():
    return render_template_string(HTML)

@app.route('/api/stats')
def stats():
    from database import db
    return jsonify(db.get_stats())

@app.route('/api/news')
def news():
    from database import db
    limit = int(request.args.get('limit', 50))
    return jsonify(db.get_news(limit=limit))

@app.route('/api/search')
def search():
    from database import db
    q = request.args.get('q', '')
    return jsonify(db.search(q, 50) if q else [])

@app.route('/api/hot')
def hot():
    from database import db
    return jsonify(db.get_top_keywords(20))

@app.route('/api/sources')
def sources():
    from database import db
    return jsonify(db.get_sources())

@app.route('/api/fetch', methods=['POST'])
def fetch():
    from async_fetcher import fetch_all
    news = fetch_all(quiet=True)
    return jsonify({'fetched': len(news)})

@app.route('/api/analyze', methods=['POST'])
def analyze():
    from ai_summarizer import summarizer
    from async_fetcher import fetch_all
    news = fetch_all(quiet=True)[:20]
    results = summarizer.analyze_batch(news)
    return jsonify({'analyzed': len(results), 'sentiment': summarizer.get_market_sentiment(results)})

def run(port=8450):
    print(f'\n  NewsHub Dashboard: http://localhost:{port}\n')
    app.run(host='0.0.0.0', port=port, debug=False)

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument('--port', type=int, default=8450)
    run(parser.parse_args().port)
