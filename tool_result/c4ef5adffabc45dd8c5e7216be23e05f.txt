def analyze_news_with_llm(title: str) -> dict:
    """使用本地LLM分析单条新闻"""
    if not USE_OLLAMA:
        return {"sentiment": "neutral", "market": "all", "urgency": "medium", "summary": ""}
    
    prompt = ANALYZE_PROMPT.format(title=title)
    response = call_ollama(prompt)
    
    if response:
        return parse_llm_response(response)
    return {"sentiment": "neutral", "market": "all", "urgency": "medium", "summary": ""}

def summarize_news_with_llm(title: str) -> str:
    """使用本地LLM生成新闻摘要"""
    if not USE_OLLAMA:
        return title[:30] + "..." if len(title) > 30 else title
    
    prompt = SUMMARIZE_PROMPT.format(title=title)
    response = call_ollama(prompt, options={"num_predict": 50})
    return response.strip() if response else title[:30] + "..."

def batch_analyze(news_list: list, delay: float = 0.5) -> list:
    """批量分析新闻"""
    if not USE_OLLAMA:
        print("  ⚠ Ollama未启用，使用规则分析")
        return news_list
    
    print("  🤖 启动本地LLM分析...")
    start = time.time()
    
    results = []
    for i, news in enumerate(news_list):
        title = news.get('title', '')
        
        # 进度显示
        if i % 5 == 0:
            print(f"    进度: {i+1}/{len(news_list)}")
        
        # LLM分析
        analysis = analyze_news_with_llm(title)
        
        # 合并结果
        news['llm_sentiment'] = analysis.get('sentiment', 'neutral')
        news['llm_market'] = analysis.get('market', 'all')
        news['llm_urgency'] = analysis.get('urgency', 'medium')
        news['llm_summary'] = analysis.get('summary', '')
        
        # 优先级调整: LLM情感优先
        if analysis.get('sentiment') in ['positive', 'negative']:
            news['sentiment'] = analysis['sentiment']
        
        results.append(news)
        time.sleep(delay)
    
    elapsed = time.time() - start
    print(f"  ✅ LLM分析完成: {len(news_list)}条 ({elapsed:.1f}s)")
    return results

def analyze_with_rules(title: str) -> dict:
    """使用规则分析新闻"""
    title_lower = title.lower()
    
    # 情感词
    positive = ['上涨', '增长', '利好', '突破', '新高', '超预期', '强劲', '反弹', 'up', 'rise', 'surge', 'gain', 'bull']
    negative = ['下跌', '暴跌', '利空', '衰退', '危机', '风险', '崩盘', 'down', 'fall', 'drop', 'crash', 'bear']
    
    sentiment = 'neutral'
    for p in positive:
        if p in title_lower:
            sentiment = 'positive'
            break
    if sentiment == 'neutral':
        for n in negative:
            if n in title_lower:
                sentiment = 'negative'
                break
    
    # 市场
    market = 'all'
    if any(w in title_lower for w in ['股票', '股', 'stock', 'index']):
        market = 'stock'
    elif any(w in title_lower for w in ['外汇', '美元', '欧元', 'forex', 'currency']):
        market = 'forex'
    elif any(w in title_lower for w in ['黄金', '原油', 'oil', 'gold', 'commodity']):
        market = 'commodity'
    
    # 紧急程度
    urgency = 'medium'
    if any(w in title_lower for w in ['战争', '核', '危机', '崩盘', 'crisis', 'war', 'crash']):
        urgency = 'high'
    elif any(w in title_lower for w in ['预计', '可能', '分析', '评论', 'may', 'might', 'could']):
        urgency = 'low'
