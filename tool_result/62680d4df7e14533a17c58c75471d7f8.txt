def cluster_news(news_list, max_clusters=5):
    hot = detect_hot_topics(news_list, top_n=max_clusters)
    clusters = []
    used = set()
    for word, count in hot:
        cluster = []
        for i, n in enumerate(news_list):
            if i in used:
                continue
            if word in n.get('title', ''):
                cluster.append(n)
                used.add(i)
        if len(cluster) >= 2:
            clusters.append({'topic': word, 'news': cluster, 'count': len(cluster)})
    return clusters

def get_market_sentiment(news_list):
    if not news_list:
        return None
    positive = sum(1 for n in news_list if n.get('llm_sentiment') == 'positive' or n.get('sentiment') == 'positive')
    negative = sum(1 for n in news_list if n.get('llm_sentiment') == 'negative' or n.get('sentiment') == 'negative')
    neutral = len(news_list) - positive - negative
    avg_score = sum(n.get('llm_score', n.get('score', 0)) for n in news_list) / len(news_list) if news_list else 0
    overall = 'bullish' if avg_score > 15 else 'bearish' if avg_score < -15 else 'neutral'
    return {
        'positive': positive,
        'negative': negative,
        'neutral': neutral,
        'avg_score': avg_score,
        'overall': overall
    }
