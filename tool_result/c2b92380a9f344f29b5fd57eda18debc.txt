class S:
    HEADER, BLUE, CYAN, GREEN, YELLOW, RED, BOLD, END = "\033[95m", "\033[94m", "\033[96m", "\033[92m", "\033[93m", "\033[91m", "\033[1m", "\033[0m"

def c(s, t): return f"{s}{t}{S.END}" if sys.stdout.isatty() else t
def B(t): return c(S.BOLD, t)
def R(t): return c(S.RED, t)
def G(t): return c(S.GREEN, t)
def Y(t): return c(S.YELLOW, t)
def L(t): return c(S.BLUE, t)
def C(t): return c(S.CYAN, t)

def info(m): print(f"  {L('▶')} {m}")
def ok(m): print(f"  {G('✓')} {G(m)}")
def warn(m): print(f"  {Y('⚠')} {Y(m)}")
def error(m): print(f"  {R('✗')} {R(m)}")

def banner(t, i="📡"):
    print()

    print(C("║") + f"  {i} {B(t)}".ljust(48) + C("║"))

def sep(): print(C("  " + "─" * 46))

def bar(cur, tot, w=25):
    f = int(w * cur / tot) if tot > 0 else 0
    return f"[{'█'*f}{'─'*(w-f)}] {cur}/{tot}"

def prog(cur, tot, pre=""):
    print(f"\r  {pre}{bar(cur, tot)}", end="", flush=True)

# 配置
def load_feishu_config():
    if CONFIG_FILE.exists():
        with open(CONFIG_FILE, 'r', encoding='utf-8') as f:
            return json.load(f)
    return {}

def load_history():
    if HISTORY_FILE.exists():
        with open(HISTORY_FILE, 'r', encoding='utf-8') as f:
            data = json.load(f)
            return data if isinstance(data, list) else data.get('sent', [])
    return []

def save_history(hist):
    with open(HISTORY_FILE, 'w', encoding='utf-8') as f:
        json.dump(hist[-1000:], f, ensure_ascii=False)

def load_log():
    if LOG_FILE.exists():
        with open(LOG_FILE, 'r', encoding='utf-8') as f:
            return json.load(f)
    return []

def save_log(logs):
    with open(LOG_FILE, 'w', encoding='utf-8') as f:
        json.dump(logs[-100:], f, ensure_ascii=False)

def add_log(mode, sent, total):
    logs = load_log()
    logs.append({"time": datetime.now().isoformat(), "mode": mode, "sent": sent, "total": total})
    save_log(logs)

# API
def get_access_token():
    cfg = load_feishu_config()
    url = "https://open.feishu.cn/open-apis/auth/v3/tenant_access_token/internal"
    data = json.dumps({"app_id": cfg['app_id'], "app_secret": cfg['app_secret']}).encode()
    req = urllib.request.Request(url, data=data, headers={'Content-Type': 'application/json'}, method='POST')
    ctx = ssl.create_default_context()
    with urllib.request.urlopen(req, timeout=10, context=ctx) as r:
        return json.loads(r.read()).get('tenant_access_token', '')

def send_card(token, receive_id, card):
    cfg = load_feishu_config()
    url = f"https://open.feishu.cn/open-apis/im/v1/messages?receive_id_type={cfg.get('receive_id_type', 'open_id')}"
    payload = json.dumps({"receive_id": receive_id, "msg_type": "interactive", "content": json.dumps(card)}).encode('utf-8')
    req = urllib.request.Request(url, data=payload, headers={'Content-Type': 'application/json', 'Authorization': f'Bearer {token}'}, method='POST')
    ctx = ssl.create_default_context()
    for _ in range(2):
        try:
            with urllib.request.urlopen(req, timeout=30, context=ctx) as r:
                return json.loads(r.read())
        except:
            time.sleep(0.5)
    return {"code": -1}

def build_news_card(title, url, source=""):
    icon = "📰" if "金十" in source else ("📊" if "Investing" in source else "📱")
    return {"header": {"title": {"tag": "plain_text", "content": f"{icon} {source}" if source else f"{icon} NewsHub"}, "template": "blue"}, "elements": [{"tag": "div", "text": {"tag": "lark_md", "content": title}}, {"tag": "hr"}, {"tag": "markdown", "content": f"[🔗 查看详情]({url})"}, {"tag": "div", "text": {"tag": "lark_md", "content": f"🕐 {datetime.now().strftime('%H:%M')}"}}]}

def build_digest_card(news_list):
    items = []
    for i, n in enumerate(news_list[:6], 1):
        title = n['title'][:30] + "..." if len(n['title']) > 30 else n['title']
        s = n.get('source', '')
        icon = "🔹" if "金十" in s else ("🔸" if "Investing" in s else "🔶")

def cmd_alert():

        time.sleep(0.2)

    add_log('alert', sent, len(items))

def cmd_category():

    banner(f"分类: {cat}", "📊")

        categories[cat_name] = categories.get(cat_name, 0) + 1
    
    info(f"分类统计: {categories}")
    
    # 筛选分类
    items = [n for n in all_news if n.get('category') == cat]
    new_items = [n for n in items if n['title'] not in history]
    
    if not new_items:

    info(f"推送: {len(new_items)} 条{cat}新闻")
    
    token = get_access_token()
    if not token:
        error("认证失败")
        return 0
    
    feishu = load_feishu_config()
    card = build_category_card(new_items[:8], cat)
    result = send_card(token, feishu.get('default_receive_id', ''), card)
    
    if result.get('code') == 0:
        for n in new_items[:8]:
            history.append(n['title'])
        save_history(history)
        ok(f"{cat}推送成功 ({len(new_items[:8])} 条)")
        add_log(f'cat_{cat}', len(new_items[:8]), len(new_items))
        return 1
    else:
        error(f"发送失败: {result.get('msg', 'Unknown')}")
        return 0

def build_summary_card(news_list):
    """构建摘要卡片"""
    # 分类统计
    cats = {}
    for n in news_list:
        cat = n.get('category', 'other')
        cats[cat] = cats.get(cat, 0) + 1
    
    # 情感统计
    sentiments = {'positive': 0, 'negative': 0, 'neutral': 0}
    for n in news_list:
        s = n.get('sentiment', 'neutral')
        sentiments[s] = sentiments.get(s, 0) + 1
    
    # 重要性统计
    imps = {'critical': 0, 'high': 0, 'medium': 0, 'low': 0}
    for n in news_list:
        imp = n.get('importance_level', 'low')
        imps[imp] = imps.get(imp, 0) + 1
    
    cat_names = {"stock": "📈股市", "forex": "💱外汇", "commodity": "🛢️商品", "macro": "🏛️宏观", "trade": "⚖️贸易", "tech": "💻科技", "china": "🇨🇳中国", "us": "🇺🇸美国", "crypto": "₿加密", "energy": "⚡能源"}
    cat_list = [f"{cat_names.get(k, k)}: {v}" for k, v in sorted(cats.items(), key=lambda x: -x[1])[:5]]
    
    return {
        "header": {"title": {"tag": "plain_text", "content": f"📊 {datetime.now().strftime('%H:%M')} 新闻摘要"}, "template": "blue"},
        "elements": [
            {"tag": "markdown", "content": f"**总新闻:** {len(news_list)} 条"},
            {"tag": "markdown", "content": f"**🟢 利好:** {sentiments['positive']} | **🔴 利空:** {sentiments['negative']} | **⚪ 中性:** {sentiments['neutral']}"},
            {"tag": "markdown", "content": f"**🔴 危机:** {imps['critical']} | **🟠 重要:** {imps['high']} | **🟡 一般:** {imps['medium']}"},
            {"tag": "hr"},
            {"tag": "markdown", "content": "**分类:** " + " | ".join(cat_list[:4])},
            {"tag": "hr"},
            {"tag": "div", "text": {"tag": "lark_md", "content": "📱 NewsHub · 自动摘要"}}
        ]
    }
