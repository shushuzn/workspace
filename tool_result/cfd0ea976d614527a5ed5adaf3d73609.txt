def cmd_alert():
    banner("关键词告警", "🔔")
    config = load_config()
    all_news = fetch_all(quiet=True)
    history = load_history()
    keywords = config.get('keywords', [])
    if not keywords:
        warn("无关键词配置")
        return 0
    new_news = [n for n in all_news if n['title'] not in history]
    items = [n for n in new_news if any(kw in n['title'] for kw in keywords)]
    if not items:
        warn("无匹配")
        return 0
    info(f"关键词: {len(keywords)} | 匹配: {len(items)}")
    token = get_access_token()
    if not token:
        error("认证失败")
        return 0
    feishu = load_feishu_config()
    sent = 0
    for i, n in enumerate(items, 1):
        prog(i, len(items), C("发送: "))
        card = build_alert_card(n['title'], n['url'], n.get('source', ''), keywords)
        result = send_card(token, feishu.get('default_receive_id', ''), card)
        if result.get('code') == 0:
            history.append(n['title'])
            ok(n['title'][:40] + "...")
            sent += 1
        time.sleep(0.2)
    print()
    save_history(history)
    add_log('alert', sent, len(items))
