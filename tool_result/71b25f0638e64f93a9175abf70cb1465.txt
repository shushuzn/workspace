def cmd_fetch(args):
    print("[INFO] Fetching news...")
    data = NewsData()
    total_new = 0
    
    for source in CONFIG["news"]["sources"]:
        print(f"  From {source['name']}...")
        if source["type"] == "rss":
            items = fetch_rss(source["url"])
        else:
            items = fetch_json(source["url"])
        
        category = source.get("category", "")
        for item in items[:CONFIG["news"]["max_per_fetch"]]:
            if data.add_news(item["title"], item["url"], source["name"], category):
                total_new += 1
                # 显示 sentiment
                sentiment, sector = analyze_sentiment(item["title"])
                print(f"    + [{sentiment}][{sector}] {item['title'][:35]}...")
    
    data.data["stats"]["last_fetch"] = datetime.now().isoformat()
    data.save()
    print(f"[OK] Fetched {total_new} new items")
