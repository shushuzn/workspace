def detect_hot_topics(news_list, top_n=5):
    """检测热点话题"""
    word_freq = {}
    
    # 中文分词简化版
    stopwords = {'的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一', '一个', '上', '也', '很', '到', '说', '要', '去', '你', '会', '着', '没有', '看', '好', '自己', '这'}
    
    for news in news_list:
        title = news.get('title', '')
        # 简单分词
        words = [w for w in re.findall(r'[\u4e00-\u9fff]{2,4}', title) if w not in stopwords]
        for w in words:
            word_freq[w] = word_freq.get(w, 0) + 1
    
    # 排序
    hot = sorted(word_freq.items(), key=lambda x: -x[1])[:top_n]
    return [(word, count) for word, count in hot if count >= 2]

def add_hot_topics(news_list):
    """为新闻列表添加热点标签"""
    hot_topics = detect_hot_topics(news_list)
    hot_words = {w for w, _ in hot_topics}
    
    for news in news_list:
        title = news.get('title', '')
        news['hot_topics'] = [w for w in hot_words if w in title]
    
    return hot_topics
