def fetch_source(key, src):
    name = src.get('name', key)
    fetch_fn = src.get('fetch_fn')
    
    if fetch_fn:
        return name, fetch_fn(src.get('limit', 10))
    elif key == 'sina':
        return name, fetch_sina_news(src.get('limit', 20))
    elif key == 'sina_stock':
        return name, fetch_sina_stock(src.get('limit', 10))
    elif key == 'eastmoney':
        return name, fetch_eastmoney(src.get('limit', 10))
    elif src.get('url'):
        return name, fetch_rss(src['url'], name)
    return name, []
