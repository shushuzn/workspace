def gen_chart(results, filename=None):
    if not PLT_AVAILABLE: return "[Chart] pip install matplotlib"
    if not filename: filename = OUTPUT / f"chart_{datetime.now().strftime('%Y%m%d_%H%M%S')}.png"
    fig, axes = plt.subplots(2, 2, figsize=(12, 8))
    syms = [r['symbol'] for r in results]
    scores = [r['score'] for r in results]
    upsides = [r['upside'] for r in results]
    colors = ['#4ade80' if s >= 65 else '#fbbf24' if s >= 50 else '#f87171' for s in scores]
    axes[0,0].barh(syms, scores, color=colors); axes[0,0].set_xlabel('Score'); axes[0,0].set_title('Score'); axes[0,0].set_xlim(0,100)
    axes[0,1].barh(syms, upsides, color=['#4ade80' if u > 0 else '#f87171' for u in upsides]); axes[0,1].set_xlabel('Upside (%)'); axes[0,1].set_title('Upside')
    axes[1,0].scatter([r['pe'] for r in results], [r['roe'] for r in results], s=80, c=scores, cmap='RdYlGn'); axes[1,0].set_xlabel('P/E'); axes[1,0].set_ylabel('ROE'); axes[1,0].set_title('P/E vs ROE'); axes[1,0].grid(True, alpha=0.3)
    ratings = {}
    for r in results:
        rt = r['rating'].split()[0]; ratings[rt] = ratings.get(rt, 0) + 1
    if ratings: axes[1,1].pie(ratings.values(), labels=ratings.keys(), autopct='%1.0f%%'); axes[1,1].set_title('Ratings')
    plt.tight_layout(); plt.savefig(filename, dpi=150, bbox_inches='tight'); plt.close()
    return f"[Chart] Generated {filename}"
