def save_db(results):
    """Save to SQLite"""
    conn = sqlite3.connect(DB_FILE)
    c = conn.cursor()
    
    c.execute('''CREATE TABLE IF NOT EXISTS stocks
                 (symbol TEXT, price REAL, target REAL, upside REAL, score INTEGER,
                  rating TEXT, pe REAL, peg REAL, roe REAL, fcf REAL, div REAL,
                  beta REAL, fetched_at TEXT, price_source TEXT)''')
    
    for r in results:
        c.execute('''INSERT INTO stocks VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?)''',
                 (r['symbol'], r['price'], r['target'], r['upside'], r['score'],
                  r['rating'], r['pe'], r['peg'], r['roe'], r['fcf'], r['div'],
                  r['beta'], r.get('fetched_at', datetime.now().isoformat()), 
                  r.get('price_source', 'unknown')))
    
    conn.commit()
    conn.close()
    return f"[DB] Saved {len(results)} stocks to {DB_FILE}"
