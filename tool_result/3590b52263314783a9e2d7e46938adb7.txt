def classify(t):
    tl = t.lower()
    for c,ks in CATEGORIES.items():

def sentiment(t):
    tl = t.lower()
    p = sum(1 for w in POSITIVE if w in tl)
    n = sum(1 for w in NEGATIVE if w in tl)
    return ("positive",20) if p>n else ("negative",-20) if n>p else ("neutral",0)

def hot_topics(news,n=5):
    sw={"的","了","在","是","和","就","不","都","一","上","也","很","到","说"}
    wf={}
    for x in news:
        for w in re.findall(r'[\u4e00-\u9fff]{2,4}',x.get("title","")):
            if w not in sw and len(w)>=2: wf[w]=wf.get(w,0)+1
    return sorted(wf.items(),key=lambda x:-x[1])[:n]

def fetch_all(quiet=False):
    c=load_cache()
    if c.get("news") and time.time()-c.get("time",0)<CACHE_TTL:
        if not quiet: print(f"  💾 缓存: {len(c['news'])} 条 ({int((time.time()-c['time'])/60)}m前)")
        return c["news"]
    results=[]; t0=time.time()
    if not quiet: print("  ▶ 并行抓取...")
    with ThreadPoolExecutor(max_workers=5) as ex:
        fs={ex.submit(f):n for n,f in SOURCES}
        for f in fs:
            try:
                ns=f.result()
                results.extend(ns)
                if not quiet: print(f"  ✓ {fs[f]}: +{len(ns)}")
            except: 
                if not quiet: print(f"  ⚠ {fs[f]}: 失败")
    save_cache({"news":results,"time":time.time()})
    if not quiet: print(f"  ⏱️ 耗时: {time.time()-t0:.1f}s")
    return results
