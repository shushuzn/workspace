def categorize(news):
    """给新闻添加分类"""
    title = news.get('title', '')
    for cat, keywords in CATEGORIES.items():
        for kw in keywords:
            if kw in title:
                news['category'] = cat
                return cat
    news['category'] = 'other'
    return 'other'

def analyze_sentiment(news):
    """分析新闻情感"""
    title = news.get('title', '')
    pos_count = sum(1 for kw in SENTIMENT_KEYWORDS['positive'] if kw in title)
    neg_count = sum(1 for kw in SENTIMENT_KEYWORDS['negative'] if kw in title)
    
    if pos_count > neg_count:
        news['sentiment'] = 'positive'
        news['sentiment_score'] = pos_count - neg_count
    elif neg_count > pos_count:
        news['sentiment'] = 'negative'
        news['sentiment_score'] = neg_count - pos_count
    else:
        news['sentiment'] = 'neutral'
        news['sentiment_score'] = 0
    return news
