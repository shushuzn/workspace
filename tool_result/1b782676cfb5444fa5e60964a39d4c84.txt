def build_alert_card(title, url, source="", keywords=None):
    """告警卡片"""
    kw_list = keywords or []
    highlight = ""
    for kw in kw_list:
        if kw in title:
            highlight = f"🔔 命中: `{kw}`"
            break
    
    return {
        "header": {
            "title": {"tag": "plain_text", "content": "⚠️ 重要告警"},
            "template": "red"
        },
        "elements": [
            {"tag": "div", "text": {"tag": "lark_md", "content": highlight}},
            {"tag": "markdown", "content": f"**{title}**"},
            {"tag": "hr"},
            {"tag": "markdown", "content": f"[🔗 查看详情]({url})"}
        ]
    }
