def cmd_stats():
    history = load_history()
    all_news = fetch_all()
    
    sources = {}
    for n in all_news:
        s = n.get('source', 'Other')
        sources[s] = sources.get(s, 0) + 1
    
    print()
    print("  ┌─ 统计 ─────────────────────────────────┐")
    print(f"  │  📰 Total: {len(all_news):<6}  📤 Sent: {len(history):<6}  │")
    print("  ├────────────────────────────────────────┤")
    for s, c in sources.items():
        print(f"  │  • {s:<14} {c:>3} 条              │")
    print("  └────────────────────────────────────────┘")
    print()
