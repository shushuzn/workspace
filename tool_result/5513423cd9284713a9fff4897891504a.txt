def cmd_digest():
    feishu = load_feishu_config()
    history = load_history()
    
    all_news = fetch_all()
    if not all_news:
        warn("No news")
        return 0
    
    new_news = [n for n in all_news if n['title'] not in history]
    items = new_news[:6]
    
    if not items:
        warn("No news")
        return 0
    
    info(f"Digest: {len(items)} items")
    print()
    
    token = get_access_token()
    if not token:
        error("Failed to get token")
        return 0
    
    try:
        card = build_digest_card(items)
        result = send_card(token, feishu.get('default_receive_id', ''), card)
        if result.get('code') == 0:
            for n in items:
                history.append(n['title'])
            save_history(history)
            ok(f"Digest sent: {len(items)} items")
            return 1
        else:
            error(f"Code {result.get('code')}")
            return 0
    except Exception as e:
        error(str(e)[:50])
        return 0
