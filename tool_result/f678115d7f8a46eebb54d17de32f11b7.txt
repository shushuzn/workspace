import time
from pathlib import Path
from datetime import datetime
from difflib import SequenceMatcher

SUBSCRIPTIONS_FILE = Path("subscriptions.json")

# 默认订阅配置

def load_subscriptions():
    """加载订阅配置"""
    if SUBSCRIPTIONS_FILE.exists():
        with open(SUBSCRIPTIONS_FILE, 'r', encoding='utf-8') as f:
            return json.load(f)
    return DEFAULT_SUBSCRIPTIONS

def save_subscriptions(subs):
    """保存订阅配置"""
    with open(SUBSCRIPTIONS_FILE, 'w', encoding='utf-8') as f:
        json.dump(subs, f, ensure_ascii=False, indent=2)

def add_subscription(keyword, level="medium", desc=""):
    """添加订阅"""
    subs = load_subscriptions()
    if "watch" not in subs:
        subs["watch"] = []
    
    for item in subs["watch"]:
        if item["keyword"].lower() == keyword.lower():
            return False
    
    subs["watch"].append({
        "keyword": keyword,
        "level": level,
        "desc": desc or keyword,
        "added_at": datetime.now().isoformat()
    })
    save_subscriptions(subs)
    return True

def remove_subscription(keyword):
    """移除订阅"""
    subs = load_subscriptions()
    if "watch" not in subs:
        return False
    
    original_len = len(subs["watch"])
    subs["watch"] = [s for s in subs["watch"] if s["keyword"].lower() != keyword.lower()]
    
    if len(subs["watch"]) < original_len:
        save_subscriptions(subs)
        return True
    return False

def match_news(news, subscriptions):
    """匹配新闻"""
    matched = []
    
    for sub in subscriptions:
        keyword = sub.get("keyword", "")
        if not keyword:
            continue
        
        for item in news:
            title = item.get("title", "").lower()
            
            if keyword.lower() in title:
                item["matched_keyword"] = keyword
                item["matched_level"] = sub.get("level", "medium")
                item["matched_desc"] = sub.get("desc", "")
                if item not in matched:
                    matched.append(item)
            elif SequenceMatcher(None, keyword.lower(), title).ratio() > 0.8:
                item["matched_keyword"] = keyword
                item["matched_level"] = sub.get("level", "medium")
                item["matched_desc"] = sub.get("desc", "")
                if item not in matched:
                    matched.append(item)
    
    return matched

def get_subscription_stats():
    """获取订阅统计"""
    subs = load_subscriptions()
    watch = subs.get("watch", [])
    
    stats = {
        "total": len(watch),
        "by_level": {"critical": 0, "high": 0, "medium": 0, "low": 0},
        "keywords": [s.get("keyword", "") for s in watch]
    }
    
    for s in watch:
        level = s.get("level", "medium")
        if level in stats["by_level"]:
            stats["by_level"][level] += 1
    
    return stats

def format_subscription_report(matched_news):
    """格式化订阅报告"""
    if not matched_news:
        return None
    
    by_level = {"critical": [], "high": [], "medium": [], "low": []}
    for n in matched_news:
        level = n.get("matched_level", "medium")
        if level in by_level:
            by_level[level].append(n)
    
    elements = [
        {"tag": "markdown", "content": f"**📬 关键词订阅** ({len(matched_news)}条匹配)"}
    ]
    
    if by_level["critical"]:
        elements.append({"tag": "hr"})
        elements.append({"tag": "markdown", "content": "### 🔴 最高级别"})
        for n in by_level["critical"][:5]:
            elements.append({"tag": "markdown", "content": f"📌 **{n.get('matched_keyword', '')}**: [{n.get('title', '')[:35]}...]({n.get('url', '')})"})
    
    if by_level["high"]:
        elements.append({"tag": "hr"})
        elements.append({"tag": "markdown", "content": "### 🟠 重要"})
        for n in by_level["high"][:5]:
            elements.append({"tag": "markdown", "content": f"📌 **{n.get('matched_keyword', '')}**: [{n.get('title', '')[:35]}...]({n.get('url', '')})"})
    
    if by_level["medium"]:
        elements.append({"tag": "hr"})
        elements.append({"tag": "markdown", "content": "### 🟡 普通"})
        for n in by_level["medium"][:5]:
            elements.append({"tag": "markdown", "content": f"📌 {n.get('matched_keyword', '')}: [{n.get('title', '')[:30]}...]({n.get('url', '')})"})
    
    return {"elements": elements}

def cmd_list():
    """列出订阅"""
    stats = get_subscription_stats()
    print("=" * 50)
    print("  [LIST] Keyword Subscriptions")
    print("=" * 50)
    print(f"\nTotal: {stats['total']} keywords\n")
    print(f"  critical: {stats['by_level']['critical']}")
    print(f"  high: {stats['by_level']['high']}")
    print(f"  medium: {stats['by_level']['medium']}")
    print(f"\nKeywords:")
    for kw in stats['keywords']:
        print(f"  - {kw}")

def cmd_check(news):
    """检查新闻"""
    subs = load_subscriptions().get("watch", [])
    matched = match_news(news, subs)
    print(f"\n[CHECK] Matched: {len(matched)}")
    
    if matched:
        by_level = {"critical": [], "high": [], "medium": [], "low": []}
        for n in matched:
            level = n.get("matched_level", "medium")
            by_level[level].append(n)
        
        for level in ["critical", "high", "medium"]:
            if by_level[level]:
                icon = "🔴" if level == "critical" else ("🟠" if level == "high" else "🟡")
                print(f"\n{icon} {level.upper()}:")
                for n in by_level[level][:3]:
                    print(f"  - [{n.get('matched_keyword', '')}] {n.get('title', '')[:40]}...")
    
    return matched

if __name__ == "__main__":
    import sys
    sys.stdout.reconfigure(encoding='utf-8')
    
    cmd = sys.argv[1] if len(sys.argv) > 1 else "list"
    
    if cmd == "list":
        cmd_list()
    elif cmd == "add" and len(sys.argv) > 2:
        keyword = sys.argv[2]
        level = sys.argv[3] if len(sys.argv) > 3 else "medium"
        if add_subscription(keyword, level):
            print(f"[OK] Added: {keyword} ({level})")
        else:
            print(f"[EXISTS] {keyword}")
    elif cmd == "remove" and len(sys.argv) > 2:
        keyword = sys.argv[2]
        if remove_subscription(keyword):
            print(f"[OK] Removed: {keyword}")
        else:
            print(f"[NOT FOUND] {keyword}")
    else:
        print("Usage: python news_subscriptions.py [list|add|remove]")
