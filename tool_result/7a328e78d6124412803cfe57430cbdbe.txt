def calc_dcf(eps_t, eps_f, beta, roic):
    near_g = min((eps_f - eps_t) / eps_t, 0.30) if eps_t > 0 else 0.15
    res = {}
    for name, gm, wm in [("Bull",1.3,0.9),("Base",1.0,1.0),("Bear",0.6,1.2)]:
        wacc = max(0.07, min(0.045 + 0.055 * beta * 0.5 * wm, 0.15))
        g_term = min(roic * 0.4, 0.035)
        pv = 0
        for y in range(1, 6):
            g = near_g * gm * (1 - y/5) + g_term * (y/5)
            g = min(max(g, 0), 0.25)
            pv += (eps_t * (1 + g) ** y) * 0.7 / (1 + wacc) ** y
        eps5 = eps_t * (1 + near_g * gm) ** 5
        tv = eps5 * (1 + g_term) * 0.7 / (wacc - g_term) if wacc > g_term else eps5 * 10
        res[name] = (pv + tv / (1 + wacc) ** 5) / 5
    return res
