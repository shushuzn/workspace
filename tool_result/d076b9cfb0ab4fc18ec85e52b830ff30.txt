        sector = news.get('sector', '通用')
        source = news.get('source', '未知')
        
        # sentiment 图标
        icon = "+" if sentiment == "利好" else "-" if sentiment == "利空" else "="
        
        # 来源简称映射
        source_short = {
            'sina_finance': '新浪财', 'sina_stock': '新浪股', 'sina_business': '新浪产',

            'bloomberg': '彭博', 'cnbc': 'CNBC', 'marketwatch': 'MW',

        }.get(source, source[:4])
        
        # 格式: 1. [+][科技][新浪财] 新闻标题...
        lines.append(f"{i}. [{icon}][{sector}][{source_short}] {news['title'][:35]}")
