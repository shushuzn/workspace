def calculate_target_price(quote: StockQuote, growth_rate: float = 0.12, pe_target: float = 30.0) -> ValuationResult:
    """计算目标价 - 基于EPS的合理估值"""
    
    # 当前EPS已经是annual
    eps_current = quote.eps if quote.eps > 0 else (quote.price / (quote.pe_ratio or 30))
    
    # 预测未来EPS
    eps_future = eps_current * (1 + growth_rate)
    
    # PE估值法：未来EPS * 目标PE
    pe_valuation = eps_future * pe_target
    
    # 简化DCF：使用Gordon Growth Model
    # 假设 FCF/EPS = 0.85 (85% conversion)
    fcf_per_share = eps_current * 0.85
    dcf_value = fcf_per_share * (1 + growth_rate) / (0.10 - 0.03)
    
    # 综合估值：PE法60% + DCF法40%
    target = pe_valuation * 0.6 + dcf_value * 0.4
    
    # 上涨空间
    upside = (target - quote.price) / quote.price * 100
    
    # 评级
    if upside > 25:
        rating = "STRONG_BUY"
    elif upside > 15:
        rating = "BUY"
    elif upside > -5:
        rating = "HOLD"
    return ValuationResult(
        pe_current=quote.pe_ratio,
        pe_target=pe_target,
        pe_valuation=round(pe_valuation, 2),
        dcf_value=round(dcf_value, 2),
        target_price=round(target, 2),
        upside_pct=round(upside, 1),
        rating=rating
    )
