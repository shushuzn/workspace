def rank_news_by_quality(news_list: List[Dict]) -> List[Dict]:
    """按质量评分排序新闻"""
    for news in news_list:
        news["quality_score"] = calculate_quality_score(news)

    return sorted(news_list, key=lambda x: x.get("quality_score", 0), reverse=True)


def get_top_quality_news(news_list: List[Dict], limit: int = 10) -> List[Dict]:
    """获取高质量新闻"""
    ranked = rank_news_by_quality(news_list)
    return ranked[:limit]
