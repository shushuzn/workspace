def export_csv(results, filename=None):
    """Export to CSV"""
    if not filename:
        filename = OUTPUT / f"stocks_{datetime.now().strftime('%Y%m%d_%H%M%S')}.csv"
    
    if not results:
        return f"[CSV] No data to export"
    
    with open(filename, 'w', newline='', encoding='utf-8') as f:
        w = csv.writer(f)
        w.writerow(["Symbol", "Price", "Target", "Upside", "Score", "Rating", 
                    "P/E", "PEG", "ROE", "FCF", "Div", "Beta", "Fetched_At", "Source"])
        for r in results:
            w.writerow([r['symbol'], r['price'], r['target'], f"{r['upside']:.1f}%",
                       r['score'], r['rating'], f"{r['pe']:.1f}", f"{r['peg']:.2f}",
                       f"{r['roe']:.1f}", f"{r['fcf']:.2f}", f"{r['div']:.2f}", r['beta'],
                       r.get('fetched_at', ''), r.get('price_source', 'unknown')])
    
    return f"[CSV] Exported {len(results)} stocks to {filename}"


def export_xlsx(results, filename=None):
    """Export to Excel (basic)"""
    try:
        import openpyxl
    except ImportError:
        return "[Excel] openpyxl not installed. Install with: pip install openpyxl"
    
    if not filename:
        filename = OUTPUT / f"stocks_{datetime.now().strftime('%Y%m%d_%H%M%S')}.xlsx"
    
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "Stocks"
    
    # Header
    headers = ["Symbol", "Price", "Target", "Upside", "Score", "Rating", 
               "P/E", "PEG", "ROE", "FCF", "Div", "Beta", "Fetched_At", "Source"]
    ws.append(headers)
    
    # Data
    for r in results:
        ws.append([r['symbol'], r['price'], r['target'], r['upside'],
                  r['score'], r['rating'], r['pe'], r['peg'],
                  r['roe'], r['fcf'], r['div'], r['beta'],
                  r.get('fetched_at', ''), r.get('price_source', '')])
    
    wb.save(filename)
    return f"[Excel] Exported {len(results)} stocks to {filename}"


def save_db(results):
    """Save to SQLite"""
    conn = sqlite3.connect(DB_FILE)
    c = conn.cursor()
    
    c.execute('''CREATE TABLE IF NOT EXISTS stocks
                 (symbol TEXT, price REAL, target REAL, upside REAL, score INTEGER,
                  rating TEXT, pe REAL, peg REAL, roe REAL, fcf REAL, div REAL,
                  beta REAL, fetched_at TEXT, price_source TEXT)''')
    
    for r in results:
        c.execute('''INSERT INTO stocks VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?)''',
                 (r['symbol'], r['price'], r['target'], r['upside'], r['score'],
                  r['rating'], r['pe'], r['peg'], r['roe'], r['fcf'], r['div'],
                  r['beta'], r.get('fetched_at', datetime.now().isoformat()), 
                  r.get('price_source', 'unknown')))
    
    conn.commit()
    conn.close()
    return f"[DB] Saved {len(results)} stocks to {DB_FILE}"

def check_alerts(results, threshold=30):
    """Check for price alerts"""
    alerts = []
    for r in results:
        if r['upside'] > threshold:
            alerts.append(f"[ALERT] {r['symbol']}: {r['upside']:+.1f}% upside (${r['price']:.2f} -> ${r['target']:.2f})")
    
    if alerts:
        return "[Alert] " + " | ".join(alerts)
    return f"[Alert] No alerts (threshold: {threshold}%)"


def gen_chart(results, filename=None):
    """Generate comparison chart"""
    if not PLT_AVAILABLE:
        return "[Chart] matplotlib not available. Install with: pip install matplotlib"
    
    if not filename:
        filename = OUTPUT / f"chart_{datetime.now().strftime('%Y%m%d_%H%M%S')}.png"
    
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))
    fig.suptitle('Stock PRO Analysis', fontsize=16, fontweight='bold')
    
    # 1. Score comparison
    ax1 = axes[0, 0]
    symbols = [r['symbol'] for r in results]
    scores = [r['score'] for r in results]
    colors = ['#4ade80' if s >= 65 else '#fbbf24' if s >= 50 else '#f87171' for s in scores]
    ax1.barh(symbols, scores, color=colors)
    ax1.set_xlabel('Score')
    ax1.set_title('Overall Score')
    ax1.set_xlim(0, 100)
    for i, v in enumerate(scores):
        ax1.text(v + 1, i, str(v), va='center')
    
    # 2. Upside comparison
    ax2 = axes[0, 1]
    upsides = [r['upside'] for r in results]
    colors2 = ['#4ade80' if u > 0 else '#f87171' for u in upsides]
    ax2.barh(symbols, upsides, color=colors2)
    ax2.set_xlabel('Upside (%)')
    ax2.set_title('Price Upside')
    ax2.axvline(x=0, color='gray', linestyle='-', linewidth=0.5)
    for i, v in enumerate(upsides):
        ax2.text(v + 1, i, f'{v:+.1f}%', va='center')
    
    # 3. P/E vs ROE
    ax3 = axes[1, 0]
    pes = [r['pe'] for r in results]
    roes = [r['roe'] for r in results]
    ax3.scatter(pes, roes, s=100, c=scores, cmap='RdYlGn', alpha=0.7)
    for i, sym in enumerate(symbols):
        ax3.annotate(sym, (pes[i], roes[i]), textcoords="offset points", xytext=(5,5), fontsize=9)
    ax3.set_xlabel('P/E Ratio')
    ax3.set_ylabel('ROE (%)')
    ax3.set_title('P/E vs ROE')
    ax3.grid(True, alpha=0.3)
    
    # 4. Rating pie
    ax4 = axes[1, 1]
    ratings = {}
    for r in results:
        rt = r['rating'].split()[0]
        ratings[rt] = ratings.get(rt, 0) + 1
    if ratings:
        ax4.pie(ratings.values(), labels=ratings.keys(), autopct='%1.0f%%', 
                colors=['#4ade80' if k == 'STRONG' else '#fbbf24' for k in ratings.keys()])
    ax4.set_title('Rating Distribution')
    
    plt.tight_layout()
    plt.savefig(filename, dpi=150, bbox_inches='tight')
    plt.close()
    
    return f"[Chart] Generated {filename}"
