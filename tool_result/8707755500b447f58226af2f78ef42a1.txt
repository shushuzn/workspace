def cmd_stats():
    history = load_history()
    all_news = fetch_all()
    
    sources = {}
    for n in all_news:
        s = n.get('source', 'Other')
        sources[s] = sources.get(s, 0) + 1
    
    print()
    print("  ┌─ 统计 ─────────────────────────────────┐")
    print(f"  │  📰 Total: {len(all_news):<6}  📤 Sent: {len(history):<6}  │")
    print("  ├────────────────────────────────────────┤")
    for s, c in sources.items():
        print(f"  │  • {s:<14} {c:>3} 条              │")
    print("  └────────────────────────────────────────┘")
    print()


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else "help"
    
    banner("NewsHub Feishu", "📡")
    
    if mode == "help":
        print("  📋 Commands:")
        print("     all     - 发送所有新消息")
        print("     alert   - 关键词告警")
        print("     digest  - 摘要 (精选6条)")
        print("     daily   - 日报统计")
        print("     test    - 测试连接")
        print("     welcome - 欢迎卡片")
        print("     stats   - 显示统计")
        print()
        return
    
    info(f"Mode: {mode}")
    info(f"Time: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    
    # 静默模式：不显示 fetcher 输出
    import sys as _sys
    from io import StringIO
    _old_stdout = _sys.stdout
    _sys.stdout = StringIO()
    
    all_news = fetch_all()
    
    _sys.stdout = _old_stdout
    
    sources = {}
    for n in all_news:
        s = n.get('source', 'Other')
        sources[s] = sources.get(s, 0) + 1
    stats(len(all_news), 0, sources)
    
    sent = 0
    if mode == "all":
        sent = cmd_all()
    elif mode == "alert":
        sent = cmd_alert()
    elif mode == "digest":
        sent = cmd_digest()
    elif mode == "daily":
        sent = cmd_daily()
    elif mode == "test":
        sent = cmd_test()
    elif mode == "welcome":
        sent = cmd_welcome()
    elif mode == "stats":
        cmd_stats()
        return
    else:
        warn(f"Unknown: {mode}")
        return
    
    print()
    sep()
    print(f"  📊 Mode: {mode}  |  📤 Sent: {sent}")
    sep()
    print()


if __name__ == "__main__":
    main()
