def cmd_digest(args):
    data = NewsData()
    days = int(args.days) if hasattr(args, 'days') else 1
    cutoff = datetime.now() - timedelta(days=days)
    recent = [n for n in data.data["news"] if datetime.fromisoformat(n.get("fetched_at","2000")).replace(tzinfo=None) > cutoff]
    
    if not recent:
        print(f"{cn('[!] No news in last ' + str(days) + ' day(s)')}")
        return
    
    now = datetime.now().strftime("%m-%d %H:%M")
    
    # 统计
    sentiments = {"利好": [], "利空": [], "中性": []}
    for n in recent:
        sentiments[n.get("sentiment","中性")].append(n)
    
    # 关键词统计
    keywords = {}
    for n in recent:
        for kw in n.get('keywords', []):
            keywords[kw] = keywords.get(kw, 0) + 1
    top_keywords = sorted(keywords.items(), key=lambda x: -x[1])[:5]
    
    # 风险评估
    risk_score = len(sentiments["利空"]) * 2 - len(sentiments["利好"])
    if risk_score > 10:
        risk_text = cn("HIGH")
    elif risk_score > 5:
        risk_text = c(Colors.YELLOW, "MEDIUM")
    else:
        risk_text = cp("LOW")
    
    total = len(recent)
    pos = len(sentiments['利好'])
    neg = len(sentiments['利空'])
    
    # 输出
    print(f"\n{c(Colors.header(), '=== DIGEST @ ' + now + ' ===')}")
    print(f"Total: {total} | {cp('+' + str(pos))} {cn('-' + str(neg))} | Risk: {risk_text}\n")
    
    # 热门标签
    if top_keywords:
        tags = " ".join([cp(f"#{k}") for k, _ in top_keywords])
        print(f"Hot: {tags}\n")
    
    # 情感详情
    for sent_type, news_list in sentiments.items():
        if not news_list:
            continue
        icon = cp("+") if sent_type == "利好" else cn("-") if sent_type == "利空" else cne("=")
        print(f"{icon} {c(Colors.BOLD, sent_type)} ({len(news_list)}):")
        for n in news_list[:2]:
            title = n.get('title', '')[:38]
            print(f"  {title}")
        if len(news_list) > 2:
            print(f"  {cne('... +' + str(len(news_list) - 2) + ' more')}")
    print()
