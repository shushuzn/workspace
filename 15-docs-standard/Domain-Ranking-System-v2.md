## 🧮 计算公式 v2.0

```python
def calculate_domain_score_v2(domain_data):
    """
    计算学科/领域段位分数 (v2.0)
    
    返回:
        total_xp: int (总经验 0-80,000,000)
        total_score: int (1-8000)
        rank: str (段位名称)
        level: int (段位内等级 1-1000)
        xp_current: int (当前段位经验)
        xp_needed: int (升级所需经验)
    """
    # 各维度得分 (0-10000)
    theory_xp = score_theory_v2(domain_data['theory'])
    tech_xp = score_technology_v2(domain_data['technology'])
    impact_xp = score_impact_v2(domain_data['impact'])
    application_xp = score_application_v2(domain_data['application'])
    talent_xp = score_talent_v2(domain_data['talent'])
    funding_xp = score_funding_v2(domain_data['funding'])
    
    # 加权总经验 (0-10000)
    weighted_xp = (
        theory_xp * 0.25 +
        tech_xp * 0.25 +
        impact_xp * 0.20 +
        application_xp * 0.15 +
        talent_xp * 0.10 +
        funding_xp * 0.05
    )
    
    # 转换为 1-8000 分
    total_score = int(weighted_xp / 10000 * 8000)
    total_score = max(1, min(8000, total_score))
    
    # 总经验
    total_xp = int(weighted_xp * 8)
    
    # 确定段位
    rank, level, xp_current, xp_needed = score_to_rank_v2(total_score)
    
    return total_xp, total_score, rank, level, xp_current, xp_needed


def score_to_rank_v2(score):
    """分数转换为段位 (v2.0: 每段位 1000 级)"""
    if score <= 1000:
        return "黑铁", score, score * 10000, 10000
    elif score <= 2000:
        return "青铜", score - 1000, (score - 1000) * 10000, 10000
    elif score <= 3000:
        return "白银", score - 2000, (score - 2000) * 10000, 10000
    elif score <= 4000:
        return "黄金", score - 3000, (score - 3000) * 10000, 10000
    elif score <= 5000:
        return "铂金", score - 4000, (score - 4000) * 10000, 10000
    elif score <= 6000:
        return "钻石", score - 5000, (score - 5000) * 10000, 10000
    elif score <= 7000:
        return "大师", score - 6000, (score - 6000) * 10000, 10000
    else:
        return "宗师", score - 7000, (score - 7000) * 10000, 10000
```
