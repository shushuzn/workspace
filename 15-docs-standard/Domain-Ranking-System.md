## 🧮 计算公式

```python
def calculate_domain_score(domain_data):
    """
    计算学科/领域段位分数
    
    参数:
        domain_data: dict, 包含各维度数据
    
    返回:
        total_score: int (1-800)
        rank: str (段位名称)
        level: int (段位内等级 1-100)
    """
    # 各维度得分 (0-100)
    theory_score = score_theory(domain_data['theory'])
    tech_score = score_technology(domain_data['technology'])
    impact_score = score_impact(domain_data['impact'])
    application_score = score_application(domain_data['application'])
    talent_score = score_talent(domain_data['talent'])
    funding_score = score_funding(domain_data['funding'])
    
    # 加权总分 (0-100)
    weighted_score = (
        theory_score * 0.25 +
        tech_score * 0.25 +
        impact_score * 0.20 +
        application_score * 0.15 +
        talent_score * 0.10 +
        funding_score * 0.05
    )
    
    # 转换为 1-800 分
    total_score = int(weighted_score * 8)
    if total_score < 1:
        total_score = 1
    elif total_score > 800:
        total_score = 800
    
    # 确定段位
    rank, level = score_to_rank(total_score)
    
    return total_score, rank, level


def score_to_rank(score):
    """分数转换为段位"""
    if score <= 100:
        return "黑铁", score
    elif score <= 200:
        return "青铜", score - 100
    elif score <= 300:
        return "白银", score - 200
    elif score <= 400:
        return "黄金", score - 300
    elif score <= 500:
        return "铂金", score - 400
    elif score <= 600:
        return "钻石", score - 500
    elif score <= 700:
        return "大师", score - 600
    else:
        return "宗师", score - 700
```
