    # VIF 检测

        issues['critical'].append(f"VIF max={vif_max:.2f} > 5")
    elif vif_max > 3:
        issues['major'].append(f"VIF max={vif_max:.2f} > 3")
    
    # Bootstrap 检测
    bootstrap_r2 = run_bootstrap(model, data, n=10000)
    if bootstrap_r2['mean'] < 0:
        issues['critical'].append(f"Bootstrap R²={bootstrap_r2['mean']:.4f} < 0")
    
    # Power 检测
    power = calculate_power(results)
    if power < 0.8:
        issues['critical'].append(f"Power={power:.4f} < 0.8")
    elif power < 0.95:
        issues['major'].append(f"Power={power:.4f} < 0.95")
    
    # ... (更多自动检测)
    
    return issues
