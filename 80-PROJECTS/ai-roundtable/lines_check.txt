74 function cleanResponse(text) {
75   text = text.replace(/\*/g, '');
76   // 清除所有括号包裹的思考内容（中英文，书名号等）
77   text = text.replace(/[（（].*?[）)]/gs, '');
78   // 清除所有thought标签残留
79   text = text.replace(/<think>.*?</think>/gs, '');
80   text = text.replace(/<think>.*$/gs, '');
81   // 清除英文括号包裹的思考
82   text = text.replace(/\(.*?\)/gs, '');
83   // 过滤指令性残留
84   text = text.replace(/[(:].*?(用户要求|用户发送|用户输入|根据规则|扮演|我需要|让我|这是一个|Style Guidance|只输出|不要|禁止|回复格式)/g, '');
85   // 过滤单行残留
86   text = text.replace(/\n[^\n]*(用户|规则|扮演|我需要|让我|回复格式)[^\n]*\n/gs, '\n');
87   // 去除空白行
88   text = text.replace(/\n{3,}/g, '\n\n');
89   text = text.replace(/^\s*\n/, '');
90   const sentences = text
91     .split(/[。\！？\n]/)
92     .map(s => s.trim())
93     .filter(s => {
94       if (s.length < 4) return false;
95       if (/^['"'']$/.test(s)) return false;
96       if (s.includes('要求我') || s.includes('扮演') || s.includes('根据规则') || s.includes('Style Guidance') || s.includes('只输出') || s.includes('不要') || s.includes('禁止') || s.includes('回复格式')) return false;
97       if (/^\(?[0-9a-zA-Z]?\s?[.、:：]?\s?(立刻|直接)/.test(s)) return false;
98       return true;
99     })
100     .slice(0, 2);
101   text = sentences.join('。').trim();
102   if (!text) return '(无有效回答)';
103   if (!/[。！？]$/.test(text)) text += '。';
104   return text;
105 }

107 // ─── 调用 ───────────────────────────────────────────
108 async function askPersona(messages, persona, topic, abortSignal, onThinking) {
109   const systemPrompt = `${persona.systemPrompt}\n\n重要：直接输出你的观点，不要使用括号、不要输出思考过程、不要输出引号、不要解释。只输出纯文本。`;
110   const allMessages = [
111     { role: 'system', name: 'system', content: systemPrompt },
112     ...messages.filter(m => m.role !== 'system'),
113   ];
114 
115   const res = await fetch(API_URL, {
116     method: 'POST',
117     headers: {
118       'Content-Type': 'application/json',
119       Authorization: `Bearer ${API_KEY}`,
120     },
121     body: JSON.stringify({
122       model: MODEL,
123       messages: allMessages,
124       max_tokens: 1500,
125       temperature: 0.2,
126       stream: false,
127     }),
128     signal: abortSignal,
129     agent,
130   });
131 
132   if (!res.ok) {
133     const err = await res.json().catch(() => ({}));
134     throw new Error(err.error?.message || `HTTP ${res.status}`);
135   }
136 
137   const data = await res.json();
138   const raw = data.choices?.[0]?.message?.content || '';
139   return cleanResponse(raw);
140 }

142 // ─── 保存结果 ───────────────────────────────────────────
143 function saveResult(topic, rounds, transcript) {
144   const timestamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
145   const safeTopic = topic.replace(/[\\/:*?"<>|]/g, '_').slice(0, 30);
146   const filename = `讨论_${safeTopic}_${timestamp}.txt`;
147   const filepath = path.join(__dirname, filename);
148 
149   const lines = [
150     '═'.repeat(60),
151     `  AI 圆桌讨论`,
152     '═'.repeat(60),
153     `话题：${topic}`,
154     `轮数：${rounds}`,
155     `时间：${new Date().toLocaleString('zh-CN')}`,
156     '─'.repeat(60),
157     '',
158     ...transcript.map(e => `${e.persona.icon} ${e.persona.name}：${e.text}`),
159     '',
160     '─'.repeat(60),
161     `共 ${transcript.length} 条发言`,
162   ];
163 
164   fs.writeFileSync(filepath, lines.join('\n'), 'utf8');
165   return filename;
166 }

182 // ─── 参数解析 ───────────────────────────────────────────
183 function parseArgs(argv) {
184   const args = argv.slice(2);
185   let topic = '';
186   let rounds = DEFAULT_ROUNDS;
187 
188   for (let i = 0; i < args.length; i++) {
189     if ((args[i] === '-r' || args[i] === '--rounds') && args[i + 1]) {
190       const n = parseInt(args[i + 1], 10);
191       if (!isNaN(n) && n > 0) rounds = Math.min(n, 10);
192       i++;
193     } else if (args[i].startsWith('-')) {
194       // ignore unknown flags
195     } else {
196       topic = args[i];
197     }
198   }
199 
200   return { topic, rounds };
201 }

203 // ─── 主函数 ───────────────────────────────────────────
204 async function main() {
205   const { topic: argTopic, rounds } = parseArgs(process.argv);
206   let topic = argTopic;
207 
208   if (!topic) {

210     const iface = rl.createInterface({ input: process.stdin, output: process.stdout });
211     topic = await new Promise(resolve => iface.question('请输入讨论话题：', resolve));
212     iface.close();
213     if (!topic.trim()) {
214       console.log('话题不能为空。');
215       process.exit(1);
216     }
217   }
218 
219   if (!API_KEY) {
220     console.error(color('错误：未设置 MINIMAX_API_KEY', 31));
221     console.error('请在 .env 文件中设置：MINIMAX_API_KEY=你的密钥');
222     process.exit(1);
223   }
224 
225   printBanner(topic, rounds);
226 
227   const abortController = new AbortController();
228   const transcript = [];
229 
230   process.on('SIGINT', () => {
231     console.log('\n\n已停止。');
232     abortController.abort();
233     process.exit(0);
234   });
235 
236   const history = [{ role: 'user', content: `话题：${topic}` }];
237 
238   try {
239     for (let round = 0; round < rounds; round++) {
240       console.log(color(`📍 第 ${round + 1} / ${rounds} 轮`, 90) + '\n');
241 
242       for (const persona of personas) {
243         const pName = color(`${persona.icon} ${persona.name}`, persona.color);
244         process.stdout.write(`  ${pName} 思考中...`);
245 
246         // 动画dots
247         let dotCount = 0;
248         const dotTimer = setInterval(() => {
249           dotCount++;
250           process.stdout.write(color('.', persona.color));
251         }, 150);
252 
253         let fullText = '';
254         try {
255           fullText = await askPersona(
256             history,
257             persona,
258             topic,
259             abortController.signal,
260           );
261         } catch (err) {
262           if (err.name === 'AbortError') {
263             clearInterval(dotTimer);
264             console.log('\n\n已停止。');
265             return;
266           }
267           fullText = color(`⚠ ${err.message}`, 31);
268         }
269 
270         clearInterval(dotTimer);
271         process.stdout.write('\r' + '\x1b[K');
272 
273         // 换行，打印最终回答
274         process.stdout.write('\r' + '\x1b[K');
275         console.log(`  ${pName}：${fullText}`);
276 
277         transcript.push({ persona, text: fullText });
278         history.push({ role: 'assistant', content: fullText });
279       }
280 
281       if (round < rounds - 1) {
282         printDivider();
283         history.push({
284           role: 'user',
285           content: `第 ${round + 2} 轮：请继续讨论，从另一角度深入。`,
286         });
287       }
288     }
289 
290     printDivider();
291     console.log(color('\n✅ 讨论结束\n', 32));
292 
293     const filename = saveResult(topic, rounds, transcript);
294     console.log(color(`💾 讨论记录已保存：${filename}`, 32));
295 
296   } catch (err) {
297     console.error(color(`\n错误：${err.message}`, 31));
298     process.exit(1);
299   }
300 }
